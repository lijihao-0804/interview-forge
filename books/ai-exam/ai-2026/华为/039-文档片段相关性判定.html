原题解析 · 华为 · 2026-09-11 · AI岗 · ACM 模式（stdin/stdout）

题目描述

将每个文档片段表示为二维点，给定邻域半径 eps 和最小点数 min_pts。一个点的邻域包含距离不超过 eps 的所有点， 包括自身 ；邻域点数至少为 min_pts 时，它是核心点。

每一步只能从核心点走到其邻域中的点。若存在某个核心点，可以分别沿这样的路径到达两个查询点，则这两个点密度相连。非核心点只能作为路径终点，不能用于连接两个核心点簇。处于核心点邻域但不是核心点的点为边界点，其他点为噪声点；噪声点与任何点（包括自身）都不密度相连。

输入描述

第一行依次为 n、m、eps、min_pts。接着 n 行，每行两个浮点坐标；随后 m 行，每行两个查询点编号。编号从 0 开始。

数据范围：1≤ n,m≤ 1000，0<eps≤ 1000，2≤ minpts≤ n，坐标均在 [-10000,10000] 内。距离采用欧氏距离。

输出描述

每个查询输出一行，密度相连为 1，否则为 0。

思路分析

第一步：建立邻域并标核心点。 两两检查距离的平方即可。为避免固定浮点容差把略大于半径的距离误判为相等，代码把输入有限十进制坐标和半径统一放大为整数，以整数平方精确比较。

第二步：只连接核心点。 两个相邻核心点可以互相到达，核心点组成的无向图可用并查集求连通分量。非核心点没有继续走出的资格，绝对不能参与合并。

第三步：给每个点维护簇集合。 核心点的集合只有自身所在分量；边界点收集邻域内全部核心点的分量，可能有多个；噪声点集合为空。

第四步：查询集合交集。 两个点密度相连，当且仅当它们的簇集合存在共同分量。遍历较小集合并在另一集合中检查成员，找到一个就可以返回。

正确性说明

合法路径的非终点都是核心点，因而某个核心点能到达的核心点恰好是其核心连通分量。它能到达的非核心点恰好邻接这一分量。两个查询点可由同一个核心点到达，等价于二者都属于该核心分量的可达点集合，即簇集合有交集。

题解代码

复杂度分析

时间复杂度 ：在定长数值运算及哈希查找均摊 O(1) 的模型下，预处理 O(n² α(n))，单次查询 O(min(s[a], s[b]))，其中 s[i] 是点 i 的簇集合大小，总计至多 O(n² α(n) + mn)。α 为反阿克曼函数。十进制缩放另需与输入数字长度相关的解析和大整数运算；任意高精度输入不能视为定长运算。

空间复杂度 ：O(n²)，邻域列表和各点簇集合；查询逐行处理，不存储全部查询。

易错点

边界点可以同时邻接两个不同核心簇，却不能把它们合并。 查询同一个噪声点仍输出 0。 半径边界使用“不超过”，重复坐标的不同点也要分别计数。

Java 实现

// 题目：文档片段相关性判定
// 算法：按题意模拟
// 模式：ACM（stdin/stdout）
// 复杂度：时间 在定长数值运算及哈希查找均摊 O(1) 的模型下，预处理 O(n² α(n))，单次查询 O(min(s[a], s[b]))，其中 s[i] 是点 i 的簇集合大小，总计至多 O(n² α(n) + mn)；空间 O(n²)，邻域列表和各点簇集合；查询逐行处理，不存储全部查询

import java.io.*;
import java.math.BigDecimal;
import java.math.BigInteger;
import java.util.*;

public class Main {
    static final char NL = (char) 10;
    // 核心方法：scaled
    static BigInteger[] scaled(String[] values) {
        int len = values.length;
        BigDecimal[] nums = new BigDecimal[len];
        int places = 0;
        // 枚举位置 i
        for (int i = 0; i < len; i++) {
            nums[i] = new BigDecimal(values[i].trim());
            int sc = nums[i].scale();
            // 条件判断
            if (sc > places) places = sc;
        }
        BigInteger[] result = new BigInteger[len];
        // 枚举位置 i
        for (int i = 0; i < len; i++) {
            BigInteger coeff = nums[i].unscaledValue();
            int exponent = -nums[i].scale();
            result[i] = coeff.multiply(BigInteger.TEN.pow(exponent + places));
        }
        // 返回结果
        return result;
    }

    // 核心方法：find
    static int find(int[] parent, int x) {
        // 循环处理
        while (parent[x] != x) {
            parent[x] = parent[parent[x]];
            x = parent[x];
        }
        // 返回结果
        return x;
    }

    // 主函数：读取输入并输出答案
    public static void main(String[] args) throws IOException {
        Scanner sc = new Scanner(System.in);
        String t = sc.next();
        // 条件判断
        if (t == null) return;
        int n = Integer.parseInt(t);
        int m = sc.nextInt();
        String epsText = sc.next();
        int minPts = sc.nextInt();
        String[] tokens = new String[1 + 2 * n];
        tokens[0] = epsText;
        // 枚举位置 i
        for (int i = 0; i < 2 * n; i++) tokens[1 + i] = sc.next();
        BigInteger[] values = scaled(tokens);
        BigInteger radius = values[0];
        BigInteger[] xs = new BigInteger[n];
        BigInteger[] ys = new BigInteger[n];
        // 枚举位置 i
        for (int i = 0; i < n; i++) {
            xs[i] = values[1 + 2 * i];
            ys[i] = values[2 + 2 * i];
        }
        BigInteger r2 = radius.multiply(radius);
        List<List<Integer>> neighbors = new ArrayList<>();
        // 枚举位置 i
        for (int i = 0; i < n; i++) neighbors.add(new ArrayList<Integer>());
        // 枚举位置 i
        for (int i = 0; i < n; i++) {
            // 枚举阶段 j
            for (int j = 0; j < n; j++) {
                BigInteger dx = xs[i].subtract(xs[j]);
                BigInteger dy = ys[i].subtract(ys[j]);
                BigInteger d2 = dx.multiply(dx).add(dy.multiply(dy));
                // 条件判断
                if (d2.compareTo(r2) <= 0) neighbors.get(i).add(j);
            }
        }
        boolean[] core = new boolean[n];
        // 枚举位置 i
        for (int i = 0; i < n; i++) core[i] = neighbors.get(i).size() >= minPts;
        int[] parent = new int[n];
        int[] size = new int[n];
        // 枚举位置 i
        for (int i = 0; i < n; i++) {
            parent[i] = i;
            size[i] = 1;
        }
        // 枚举位置 i
        for (int i = 0; i < n; i++) {
            // 条件判断
            if (!core[i]) continue;
            // 枚举阶段 j
            for (int j : neighbors.get(i)) {
                // 条件判断
                if (core[j]) {
                    int a = find(parent, i);
                    int b = find(parent, j);
                    // 条件判断
                    if (a != b) {
                        // 条件判断
                        if (size[a] < size[b]) {
                            int tmp = a;
                            a = b;
                            b = tmp;
                        }
                        parent[b] = a;
                        size[a] += size[b];
                    }
                }
            }
        }
        List<Set<Integer>> memberships = new ArrayList<>();
        // 枚举位置 i
        for (int i = 0; i < n; i++) {
            Set<Integer> set = new HashSet<>();
            // 枚举阶段 j
            for (int j : neighbors.get(i)) {
                // 条件判断
                if (core[j]) set.add(find(parent, j));
            }
            memberships.add(set);
        }
        StringBuilder sb = new StringBuilder();
        // 循环处理
        for (int q = 0; q < m; q++) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            Set<Integer> first = memberships.get(a);
            Set<Integer> second = memberships.get(b);
            // 条件判断
            if (first.size() > second.size()) {
                Set<Integer> tmp = first;
                first = second;
                second = tmp;
            }
            int ans = 0;
            // 循环处理
            for (int root : first) {
                // 条件判断
                if (second.contains(root)) {
                    ans = 1;
                    break;
                }
            }
            // 条件判断
            if (q > 0) sb.append(NL);
            sb.append(ans);
        }
        System.out.print(sb);
    }
}

Python 实现

# 题目：文档片段相关性判定
# 算法：按题意模拟
# 模式：ACM（stdin/stdout）
# 复杂度：时间 在定长数值运算及哈希查找均摊 O(1) 的模型下，预处理 O(n² α(n))，单次查询 O(min(s[a], s[b]))，其中 s[i] 是点 i 的簇集合大小，总计至多 O(n² α(n) + mn)；空间 O(n²)，邻域列表和各点簇集合；查询逐行处理，不存储全部查询

import sys
from decimal import Decimal


# 核心函数：scaled
def scaled(values):
    numbers = [Decimal(value) for value in values]
    places = max(0, max(-value.as_tuple().exponent for value in numbers))
    result = []
    # 遍历 value
    for value in numbers:
        sign, digits, exponent = value.as_tuple()
        coefficient = 0
        # 遍历 digit
        for digit in digits:
            coefficient = coefficient * 10 + digit
        result.append((-1 if sign else 1) * coefficient * 10 ** (exponent + places))
    # 返回结果
    return result


# 核心函数：cluster_memberships
def cluster_memberships(points, radius, min_pts):
    n = len(points)
    neighbors = [[] for _ in range(n)]
    for i, (x, y) in enumerate(points):
        for j, (u, v) in enumerate(points):
            # 条件判断
            if (x - u) ** 2 + (y - v) ** 2 <= radius ** 2:
                neighbors[i].append(j)
    core = [len(row) >= min_pts for row in neighbors]
    parent = list(range(n))
    size = [1] * n

    # 核心函数：find
    def find(x):
        # 循环处理
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        # 返回结果
        return x

    # 枚举位置 i
    for i in range(n):
        # 条件判断
        if not core[i]:
            continue
        # 枚举阶段 j
        for j in neighbors[i]:
            # 条件判断
            if core[j]:
                a, b = find(i), find(j)
                # 条件判断
                if a != b:
                    # 条件判断
                    if size[a] < size[b]:
                        a, b = b, a
                    parent[b] = a
                    size[a] += size[b]
    # 返回结果
    return [{find(j) for j in row if core[j]} for row in neighbors]


# 读取输入并输出答案
def solve():
    input = sys.stdin.readline
    n, m, eps, min_pts = input().split()
    n, m, min_pts = int(n), int(m), int(min_pts)
    tokens = [eps]
    # 遍历 _
    for _ in range(n):
        tokens.extend(input().split())
    values = scaled(tokens)
    points = list(zip(values[1::2], values[2::2]))
    memberships = cluster_memberships(points, values[0], min_pts)
    # 遍历 _
    for _ in range(m):
        a, b = map(int, input().split())
        first, second = memberships[a], memberships[b]
        # 条件判断
        if len(first) > len(second):
            first, second = second, first
        print(int(any(root in second for root in first)))


if __name__ == '__main__':
    solve()

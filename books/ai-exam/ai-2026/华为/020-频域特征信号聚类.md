# 频域特征信号聚类

> <a href="https://onefly.top/zero2Leetcode/04_real_interviews/huawei/ai-20260617/" target="_blank" rel="noopener noreferrer">原题解析</a> · 华为 · 2026-06-17 · AI岗 · ACM 模式（stdin/stdout）

## 题目描述

设计一个基于频域特征的 K-Means 聚类系统，三步流水线：

DFT 频域变换 ：对 $N$ 组长度为 $L$ 的信号做离散傅里叶变换 提取特征 ：从 $k = 1$ 到 $L/2 - 1$ 的频点中，取幅值最大的 3 个频率索引 $k$ 作为特征向量（按幅值降序，幅值相同时索引小的优先） K-Means 聚类 ：给定 $K$ 个初始中心索引，使用欧氏距离做分配，均值用银行家舍入取整作为新中心，空簇保留原中心，最多迭代 100 次或中心不变时停止

输出收敛后的 $K$ 个聚类中心，按字典序排序。

## 思路分析

第一步：DFT 频域变换

对每组信号 $x[n]$，计算 $X[k] = \sum_{n=0}^{L-1} x[n] \cdot e^{-j2\pi kn/L}$。展开为实部 $\text{Re}[k] = \sum x[n] \cos(2\pi kn/L)$ 和虚部 $\text{Im}[k] = -\sum x[n] \sin(2\pi kn/L)$。

用 NumPy 向量化：构建角度矩阵后一次矩阵乘法得到所有频点的实部和虚部。

第二步：提取频域特征

幅值 $\lvert X[k] \rvert = \sqrt{\text{Re}[k]^2 + \text{Im}[k]^2}$。只看 $k = 1$ 到 $L/2 - 1$（排除直流分量 $k=0$ 和奈奎斯特频率 $k=L/2$）。按幅值降序排序取前 3 个频率索引作为特征向量。

第三步：K-Means 聚类细节

分配阶段 ：用平方欧氏距离（整数运算精确）比远近 更新阶段 ：银行家舍入（四舍六入五成双）——用整数算术判断余数与分母的关系，恰好一半时凑偶数 空簇处理 ：保持原中心不变 收敛判定 ：中心都是整数，前后完全相同即收敛

银行家舍入的整数实现：设 $\text{sum}/\text{cnt}$ 得商 $q$ 余 $r$，若 $2r < \text{cnt}$ 舍去；$2r > \text{cnt}$ 进位；$2r = \text{cnt}$ 时 $q$ 为偶则舍、为奇则进。

## 题解代码

### Python 实现

```python
# 题目：频域特征信号聚类
# 算法：排序 + 扫描
# 模式：ACM（stdin/stdout）
# 复杂度：时间 DFT 为 O(N L^2)，K-Means 每轮 O(N K) 最多 100 轮，总计 O(N L^2 + 100 N K)；空间 O(N L) 存信号和角度矩阵

import sys
import numpy as np


# 核心函数：banker_round
def banker_round(num, den):
    q, r = divmod(num, den)
    twice = 2 * r
    # 条件判断
    if twice < den:
        # 返回结果
        return q
    # 条件判断
    if twice > den:
        # 返回结果
        return q + 1
    # 返回结果
    return q if q % 2 == 0 else q + 1


# 核心函数：feature_of
def feature_of(signal, L):
    x = np.asarray(signal, dtype=float)
    kmax = L // 2 - 1
    ks = np.arange(1, kmax + 1)
    n = np.arange(L)
    ang = 2.0 * np.pi * np.outer(ks, n) / L
    re = (x * np.cos(ang)).sum(axis=1)
    im = -(x * np.sin(ang)).sum(axis=1)
    mag = np.sqrt(re * re + im * im)
    order = sorted(range(len(ks)), key=lambda i: (-mag[i], int(ks[i])))
    # 返回结果
    return [int(ks[order[0]]), int(ks[order[1]]), int(ks[order[2]])]


# 核心函数：sq_dist
def sq_dist(a, b):
    # 返回结果
    return sum((a[t] - b[t]) ** 2 for t in range(3))


# 核心函数：main
def main():
    data = sys.stdin.read().split()
    idx = 0
    N = int(data[idx]); L = int(data[idx+1]); K = int(data[idx+2]); idx += 3
    signals = []
    # 遍历 _
    for _ in range(N):
        row = [float(data[idx + j]) for j in range(L)]
        idx += L
        signals.append(row)
    init = [int(data[idx + j]) for j in range(K)]

    features = [feature_of(sig, L) for sig in signals]
    centers = [features[i][:] for i in init]

    # 遍历 _
    for _ in range(100):
        clusters = [[] for _ in range(K)]
        # 遍历 f
        for f in features:
            best, best_d = 0, None
            # 枚举阶段 j
            for j in range(K):
                d = sq_dist(f, centers[j])
                # 条件判断
                if best_d is None or d < best_d:
                    best_d, best = d, j
            clusters[best].append(f)
        new_centers = []
        # 枚举阶段 j
        for j in range(K):
            # 条件判断
            if not clusters[j]:
                new_centers.append(centers[j][:])
                continue
            cnt = len(clusters[j])
            nc = [banker_round(sum(f[t] for f in clusters[j]), cnt) for t in range(3)]
            new_centers.append(nc)
        # 条件判断
        if new_centers == centers:
            break
        centers = new_centers

    centers_sorted = sorted(centers)
    print("\n".join(f"{c[0]} {c[1]} {c[2]}" for c in centers_sorted))


if __name__ == "__main__":
    main()
```

## 复杂度分析

时间复杂度 ：DFT 为 $O(N \times L^2)$，K-Means 每轮 $O(N \times K)$ 最多 100 轮，总计 $O(N \times L^2 + 100 \times N \times K)$。 空间复杂度 ：$O(N \times L)$ 存信号和角度矩阵。

### Java 实现

```java
// 题目：频域特征信号聚类
// 算法：排序 + 扫描
// 模式：ACM（stdin/stdout）
// 复杂度：时间 DFT 为 O(N L^2)，K-Means 每轮 O(N K) 最多 100 轮，总计 O(N L^2 + 100 N K)；空间 O(N L) 存信号和角度矩阵

import java.io.*;
import java.util.*;

public class Main {
    static BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
    static StringTokenizer st;
    static String next() throws IOException {
        while (st == null || !st.hasMoreTokens()) {
            String line = br.readLine();
            if (line == null) return null;
            st = new StringTokenizer(line);
        }
        return st.nextToken();
    }
    static int nextInt() throws IOException { return Integer.parseInt(next()); }
    static long nextLong() throws IOException { return Long.parseLong(next()); }

    // 核心方法：featureOf
    static int[] featureOf(double[] signal, int L) {
        int m = L / 2 - 1;
        int[] ks = new int[m];
        double[] mag = new double[m];
        // 循环处理
        for (int idx = 0; idx < m; idx++) {
            int kk = idx + 1; ks[idx] = kk;
            double re = 0, im = 0;
            // 循环处理
            for (int nn = 0; nn < L; nn++) { double ang = 2.0 * Math.PI * kk * nn / L; re += signal[nn] * Math.cos(ang); im += -signal[nn] * Math.sin(ang); }
            mag[idx] = Math.sqrt(re * re + im * im);
        }
        Integer[] ord = new Integer[m];
        // 枚举位置 i
        for (int i = 0; i < m; i++) ord[i] = i;
        final double[] mg = mag; final int[] kks = ks;
        Arrays.sort(ord, (a, b) -> { int c = Double.compare(mg[b], mg[a]); if (c != 0) return c; return Integer.compare(kks[a], kks[b]); });
        // 返回结果
        return new int[]{ks[ord[0]], ks[ord[1]], ks[ord[2]]};
    }
    // 核心方法：sq
    static long sq(int[] a, int[] b) { long s = 0; for (int t = 0; t < 3; t++) { long d = (long) a[t] - b[t]; s += d * d; } return s; }
    // 核心方法：bankerRound
    static long bankerRound(long num, long den) { long q = num / den; long r = num % den; long twice = 2 * r; if (twice < den) return q; if (twice > den) return q + 1; return (q % 2 == 0) ? q : q + 1; }
    // 主函数：读取输入并输出答案
    public static void main(String[] args) throws IOException {
        int N = nextInt(), L = nextInt(), K = nextInt();
        double[][] signals = new double[N][L];
        // 枚举位置 i
        for (int i = 0; i < N; i++) for (int j = 0; j < L; j++) signals[i][j] = Double.parseDouble(next());
        int[] init = new int[K];
        // 枚举位置 i
        for (int i = 0; i < K; i++) init[i] = nextInt();
        int[][] features = new int[N][3];
        // 枚举位置 i
        for (int i = 0; i < N; i++) features[i] = featureOf(signals[i], L);
        int[][] centers = new int[K][3];
        // 循环处理
        for (int c = 0; c < K; c++) centers[c] = features[init[c]].clone();
        // 循环处理
        for (int iter = 0; iter < 100; iter++) {
            List<List<Integer>> clusters = new ArrayList<>();
            // 循环处理
            for (int c = 0; c < K; c++) clusters.add(new ArrayList<>());
            // 枚举位置 i
            for (int i = 0; i < N; i++) { int best = 0; long bd = Long.MAX_VALUE; for (int c = 0; c < K; c++) { long d = sq(features[i], centers[c]); if (d < bd) { bd = d; best = c; } } clusters.get(best).add(i); }
            int[][] nc = new int[K][3];
            // 循环处理
            for (int c = 0; c < K; c++) {
                List<Integer> cl = clusters.get(c);
                // 条件判断
                if (cl.isEmpty()) { nc[c] = centers[c].clone(); continue; }
                long cnt = cl.size();
                // 循环处理
                for (int t = 0; t < 3; t++) { long sum = 0; for (int idx : cl) sum += features[idx][t]; nc[c][t] = (int) bankerRound(sum, cnt); }
            }
            boolean same = true;
            // 循环处理
            for (int c = 0; c < K; c++) for (int t = 0; t < 3; t++) if (nc[c][t] != centers[c][t]) same = false;
            // 条件判断
            if (same) break;
            centers = nc;
        }
        int[][] sorted = centers.clone();
        Arrays.sort(sorted, (x, y) -> { if (x[0] != y[0]) return Integer.compare(x[0], y[0]); if (x[1] != y[1]) return Integer.compare(x[1], y[1]); return Integer.compare(x[2], y[2]); });
        StringBuilder sb = new StringBuilder();
        // 循环处理
        for (int c = 0; c < K; c++) { if (c > 0) sb.append((char)10); sb.append(sorted[c][0]).append(' ').append(sorted[c][1]).append(' ').append(sorted[c][2]); }
        System.out.println(sb);
    }
}
```

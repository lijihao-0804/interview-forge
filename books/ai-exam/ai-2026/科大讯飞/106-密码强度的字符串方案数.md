```text
## 密码强度的字符串方案数 - 线性 DP

MOD = 10 ** 9 + 7


def build_answers(maxn):
    """f[j][i]：长度为 i、结尾一段相同字符恰好连续 j 个（j=1,2,3）的方案数。
    返回 res[i] = 长度 i 的合法串总数，多组询问共用这张表。"""
    f = [[0] * (maxn + 1) for _ in range(4)]
    res = [0] * (maxn + 1)
    # 长度为 1 时结尾游程必然是 1，26 个字母各算一种
    f[1][1] = 26 % MOD
    res[1] = f[1][1]
    for i in range(2, maxn + 1):
        # 新字符与前一个不同：前一位是什么结尾状态都行，换成另外 25 个字母之一
        f[1][i] = 25 * (f[1][i - 1] + f[2][i - 1] + f[3][i - 1]) % MOD
        # 新字符与前一个相同：结尾游程加长一节，所以只能由长度小 1 的状态转来
        f[2][i] = f[1][i - 1]
        # 游程到 3 就封顶，f[3] 再接同字符就是 4 连，直接不转移，天然满足限制
        f[3][i] = f[2][i - 1]
        res[i] = (f[1][i] + f[2][i] + f[3][i]) % MOD
    return res


## 先把全部询问读进来，才能只预处理到实际用到的最大长度
t = int(input())
queries = [int(input()) for _ in range(t)]
answers = build_answers(max(queries))
## 表建好后每组询问只是一次下标访问，逐行输出即可
for n in queries:
    print(answers[n])
```

### Java 实现

```java
// 题目：密码强度的字符串方案数
// 算法：按题意模拟
// 模式：ACM（stdin/stdout）

import java.io.*;
import java.util.*;

public class Main {
    static final long MOD = 1000000007L;

    static long[] buildAnswers(int maxn) {
        long[][] f = new long[4][maxn + 1];
        long[] res = new long[maxn + 1];
        f[1][1] = 26 % MOD;
        res[1] = f[1][1];
        for (int i = 2; i <= maxn; i++) {
            f[1][i] = 25 * ((f[1][i - 1] + f[2][i - 1] + f[3][i - 1]) % MOD) % MOD;
            f[2][i] = f[1][i - 1];
            f[3][i] = f[2][i - 1];
            res[i] = (f[1][i] + f[2][i] + f[3][i]) % MOD;
        }
        return res;
    }

    // 主函数：读取输入并输出答案
    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String line = br.readLine();
        while (line != null && line.trim().isEmpty()) line = br.readLine();
        int t = Integer.parseInt(line.trim());
        int[] queries = new int[t];
        int maxn = 1;
        for (int i = 0; i < t; i++) {
            String s = br.readLine();
            while (s != null && s.trim().isEmpty()) s = br.readLine();
            queries[i] = Integer.parseInt(s.trim());
            if (queries[i] > maxn) maxn = queries[i];
        }
        long[] answers = buildAnswers(maxn);
        for (int n : queries) System.out.println(answers[n]);
    }
}
```

正确性说明

从左向右处理首个不足目标的位置，任何方案都必须补足该缺口。把覆盖它的操作尽量右移，既不破坏已满足位置，又不减少对未来位置的覆盖，故贪心使用最少操作。差分只压缩相同的区间更新；单调判定上的二分得到最大可行目标。

易错点与边界

数组允许负数，k 可以为 0；末尾起点需要截到 n-w，不能创建越界窗口。

import java.io.*;
import java.util.*;

public class Main {
 // 核心方法：check
 static boolean check(long[] a,int n,int w,long k,long target){
  long[] expire=new long[n+1];int last=n-w;long add=0;long used=0;
  // 枚举位置 i
  for(int i=0;i<n;i++){
   add-=expire[i];
   long need=target-a[i]-add;
   // 条件判断
   if(need>0){used+=need;if(used>k)return false;add+=need;int start=i<last?i:last;expire[start+w]+=need;}
  }
  // 返回结果
  return true;
 }
 // 核心方法：maxMinValue
 static long maxMinValue(long[] a,int n,int w,long k){
  long lo=a[0];for(int i=1;i<n;i++)if(a[i]<lo)lo=a[i];
  long hi=lo+k;
  // 循环处理
  while(lo<hi){long mid=lo+(hi-lo+1)/2;if(check(a,n,w,k,mid))lo=mid;else hi=mid-1;}

  return lo;

Python 实现

# 题目：定长区间加一最大化最小值
# 算法：按题意模拟
# 模式：ACM（stdin/stdout）
# 复杂度：时间 O(n log(k+2))，包含 k=0 时的输入和最小值扫描；空间 O(n)

import sys
input = sys.stdin.readline

# 核心函数：check
def check(a, n, w, k, target):
    expire = [0] * (n + 1)
    last = n - w  # 窗口起点（0 下标）最多只能到 n-w
    add = 0       # 当前位置被所有生效窗口加了多少
    used = 0      # 已经用掉的操作次数
    # 枚举位置 i
    for i in range(n):
        add -= expire[i]
        need = target - a[i] - add
        # 条件判断
        if need > 0:
            used += need
            # 条件判断
            if used > k:
                # 返回结果
                return False
            add += need
            start = i if i < last else last
            expire[start + w] += need
    # 返回结果
    return True

# 核心函数：max_min_value
def max_min_value(a, n, w, k):
    lo = min(a)
    hi = lo + k
    # 循环处理
    while lo < hi:
        mid = lo + (hi - lo + 1) // 2
        # 条件判断
        if check(a, n, w, k, mid):
            lo = mid
        else:
            hi = mid - 1
    # 返回结果
    return lo

n, w, k = map(int, input().split())
a = list(map(int, input().split()))
print(max_min_value(a, n, w, k))

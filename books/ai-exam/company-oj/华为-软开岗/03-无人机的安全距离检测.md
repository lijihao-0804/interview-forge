# 无人机的安全距离检测

> 来源：语雀「华为-软开岗真题题解」（已做 OCR 校对）

## 题目描述

在某重大活动现场，无人机将在夜空中展示一幅灯光图案。为了保证飞行安全，需要对无人机之间的安全距离进行检测。

现在已经根据灯光设计给出所有无人机的固定坐标 (X, Y)。需要设计一个系统，根据输入的无人机坐标，计算出这些无人机之间的最小欧氏距离。

欧氏距离即平面上两点的直线距离：对两点 (X1, Y1) 与 (X2, Y2)，欧氏距离为 √[(X2−X1)² + (Y2−Y1)²]。

## 输入描述

第一行为无人机的数量 N（0 < N < 50000）。

后面 N 行为 N 架无人机的坐标 X 和 Y，X、Y 均为整数，且 0 < X, Y < 10000。

输入示例（原文如此，每行两个坐标，OCR 丢失了空格）：
```
5
1 1
10 18
17 50
22 25
39 50
```

## 输出描述

输出这些无人机之间的最小欧氏距离。（原文输出格式及精度要求模糊，未明确说明）

## 解题思路

本题是经典的「平面最近点对」问题，求 N 个点之间的最小欧氏距离。
采用分治：先按 x 排序，递归处理左右两半并得到各自的最小距离 best。
合并时只需检查与中线 x 距离平方小于 best 的点构成「条带」，把条带按 y 排序后，每个点只与后面 y 差平方小于 best 的点比较（最多常数个），即可 O(n) 完成合并。
递归结束后对 best（距离的平方，用 long 避免溢出）开方输出，保留四位小数。
全程用距离平方比较，避免反复开方带来的精度误差与性能开销。

## 复杂度分析

- 时间复杂度：O(n log² n)，分治每层对条带排序 O(n log n)，共 O(log n) 层（用归并优化可到 O(n log n)）。
- 空间复杂度：O(n)，递归栈与条带、归并临时数组。

### Java 实现

```java
import java.io.*;
import java.util.*;

public class Main {
    static class Pt {
        long x, y;
        Pt(long x, long y) { this.x = x; this.y = y; }
    }

    static Pt[] ps;
    static long best;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String line = br.readLine();
        if (line == null) return;
        int n = Integer.parseInt(line.trim());
        ps = new Pt[n];
        for (int i = 0; i < n; i++) {
            StringTokenizer st = new StringTokenizer(br.readLine());
            ps[i] = new Pt(Long.parseLong(st.nextToken()), Long.parseLong(st.nextToken()));
        }
        if (n < 2) { System.out.println("0.0000"); return; }
        Arrays.sort(ps, (a, b) -> a.x != b.x ? Long.compare(a.x, b.x) : Long.compare(a.y, b.y));
        best = Long.MAX_VALUE;
        rec(0, n - 1);
        System.out.printf(Locale.ROOT, "%.4f%n", Math.sqrt((double) best));
    }

    static long d2(Pt a, Pt b) {
        long dx = a.x - b.x, dy = a.y - b.y;
        return dx * dx + dy * dy;
    }

    static void rec(int l, int r) {
        if (r - l <= 3) {
            for (int i = l; i <= r; i++)
                for (int j = i + 1; j <= r; j++) {
                    long d = d2(ps[i], ps[j]);
                    if (d < best) best = d;
                }
            Arrays.sort(ps, l, r + 1, (a, b) -> Long.compare(a.y, b.y));
            return;
        }
        int mid = (l + r) >>> 1;
        long midx = ps[mid].x;
        rec(l, mid);
        rec(mid + 1, r);
        ArrayList<Pt> strip = new ArrayList<>();
        for (int i = l; i <= r; i++) {
            long dx = ps[i].x - midx;
            if (dx * dx < best) strip.add(ps[i]);
        }
        strip.sort((a, b) -> Long.compare(a.y, b.y));
        for (int i = 0; i < strip.size(); i++) {
            for (int j = i + 1; j < strip.size(); j++) {
                long dy = strip.get(j).y - strip.get(i).y;
                if (dy * dy >= best) break;
                long d = d2(strip.get(i), strip.get(j));
                if (d < best) best = d;
            }
        }
        Pt[] merged = new Pt[r - l + 1];
        int i = l, j = mid + 1, k = 0;
        while (i <= mid && j <= r) {
            if (ps[i].y <= ps[j].y) merged[k++] = ps[i++]; else merged[k++] = ps[j++];
        }
        while (i <= mid) merged[k++] = ps[i++];
        while (j <= r) merged[k++] = ps[j++];
        System.arraycopy(merged, 0, ps, l, merged.length);
    }
}
```


### Python 实现

```python
import sys
import math

def main():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    ps = []
    for _ in range(n):
        x = int(next(it))
        y = int(next(it))
        ps.append((x, y))
    if n < 2:
        print('0.0000')
        return
    ps.sort(key=lambda p: (p[0], p[1]))
    best = [float('inf')]

    def d2(a, b):
        dx = a[0] - b[0]
        dy = a[1] - b[1]
        return dx * dx + dy * dy

    def rec(l, r):
        if r - l <= 3:
            for i in range(l, r + 1):
                for j in range(i + 1, r + 1):
                    d = d2(ps[i], ps[j])
                    if d < best[0]:
                        best[0] = d
            ps[l:r + 1] = sorted(ps[l:r + 1], key=lambda p: p[1])
            return
        mid = (l + r) // 2
        midx = ps[mid][0]
        rec(l, mid)
        rec(mid + 1, r)
        strip = []
        for i in range(l, r + 1):
            dx = ps[i][0] - midx
            if dx * dx < best[0]:
                strip.append(ps[i])
        strip.sort(key=lambda p: p[1])
        for i in range(len(strip)):
            for j in range(i + 1, len(strip)):
                dy = strip[j][1] - strip[i][1]
                if dy * dy >= best[0]:
                    break
                d = d2(strip[i], strip[j])
                if d < best[0]:
                    best[0] = d
        merged = []
        i = l
        j = mid + 1
        while i <= mid and j <= r:
            if ps[i][1] <= ps[j][1]:
                merged.append(ps[i])
                i += 1
            else:
                merged.append(ps[j])
                j += 1
        while i <= mid:
            merged.append(ps[i])
            i += 1
        while j <= r:
            merged.append(ps[j])
            j += 1
        ps[l:r + 1] = merged

    rec(0, n - 1)
    print('%.4f' % math.sqrt(best[0]))

main()
```

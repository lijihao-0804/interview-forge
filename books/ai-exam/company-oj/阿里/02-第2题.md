# 第2题

> 来源：语雀「阿里真题题解」（已做 OCR 校对）

## 题目描述

在一条数轴上有 N 个机器人与 M 个充电桩，机器人与充电桩的位置均为整数。第 j 个充电桩可同时为至多 C_j 个机器人提供充电（容量可被多个机器人占用，彼此独立）。若某机器人与某充电桩的距离（两者位置差的绝对值）不超过某个整数阈值 E，则视为该机器人可以到达该充电桩充电。

请你最小化阈值 E，使得所有 N 个机器人都能被分配到某一充电桩，且不超过对应充电桩的容量限制。若无论如何分配都无法覆盖全部机器人，则输出 -1。

## 输入描述

每个测试文件包含多组测试数据。第一行输入整数 T（1 ≤ T ≤ 10）表示测试组数。每组数据格式如下：
- 第一行输入两个整数 N, M；（原文取值范围识别模糊）
- 第二行输入 N 个整数，表示 N 个机器人（扫地机器人）的位置；
- 第三行输入 M 个整数，表示 M 个充电桩的位置；
- 第四行输入 M 个整数 C_1, …, C_M，表示各充电桩的容量。

（原文取值范围识别严重缺失，仅大意为各位置与容量为非负整数。）

## 输出描述

对于每一组测试数据，新起一行。若无法覆盖全部机器人则输出 -1，否则输出一个整数，表示最小的阈值 E。

## 示例 1

输入：
```
2
2 2
1 5
2 6
1 1
3 2
0 10 20
5 15
1 1
```

输出：
```
1
-1
```

## 说明

第一组：取 E=1，可分配机器人 1→桩 2、机器人 5→桩 6，可行；无法取更小，故答案为 1。

第二组：所有充电桩总容量为 2 < 3（机器人数），无论 E 取何值都无法覆盖全部机器人，输出 -1。

（原文第二组数据首行 N、M 的数值缺失，此处“3 2”系据上下文推断；其余位置与容量数值为原文保留。）

## 解题思路

本题要最小化最大距离阈值 E，使每个机器人都能被分配到某个距离 ≤ E 的充电桩且不超过容量——典型的「二分答案 + 最大流可行性判定」。
可行性判定：建二分图最大流网络，源点连每个机器人（容量 1），机器人在距离 ≤ E 时连对应充电桩（容量 1），每个充电桩连汇点（容量 C_j）；当最大流等于机器人数 nR 时该 E 可行（用 Dinic 求解，达到 nR 即提前返回）。
把所有可能的距离（机器人与充电桩两两距离，外加 0）排序去重，在其上二分查找最小的可行 E。
若所有充电桩总容量都小于机器人数，直接输出 −1。

## 复杂度分析

- 时间复杂度：O(log(NM) · MaxFlow)，二分 O(log(NM)) 次，每次在 O(N+M+NM) 规模的网络上跑 Dinic 最大流。
- 空间复杂度：O(N+M+NM)，存图与距离候选。

## 解题代码

### Java 实现

```java
// 题目：第2题
// 算法：按题意模拟
// 模式：ACM（stdin/stdout）

import java.io.*;
import java.util.*;

public class Main {
    static int nR, mS;
    static long[] rpos, spos;
    static int[] scap;
    static List<Edge>[] g;
    static int[] level, iter;

    static class Edge {
        int to, rev, cap;
        Edge(int to, int rev, int cap) { this.to = to; this.rev = rev; this.cap = cap; }
    }

    static void addEdge(int f, int t, int c) {
        g[f].add(new Edge(t, g[t].size(), c));
        g[t].add(new Edge(f, g[f].size() - 1, 0));
    }

    static boolean bfs(int s, int t) {
        Arrays.fill(level, -1);
        ArrayDeque<Integer> q = new ArrayDeque<>();
        level[s] = 0; q.add(s);
        while (!q.isEmpty()) {
            int v = q.poll();
            for (Edge e : g[v]) {
                if (e.cap > 0 && level[e.to] < 0) {
                    level[e.to] = level[v] + 1;
                    q.add(e.to);
                }
            }
        }
        return level[t] >= 0;
    }

    static int dfs(int v, int t, int f) {
        if (v == t) return f;
        for (; iter[v] < g[v].size(); iter[v]++) {
            Edge e = g[v].get(iter[v]);
            if (e.cap > 0 && level[v] + 1 == level[e.to]) {
                int d = dfs(e.to, t, Math.min(f, e.cap));
                if (d > 0) { e.cap -= d; g[e.to].get(e.rev).cap += d; return d; }
            }
        }
        return 0;
    }

    static boolean feasible(long E) {
        int nodes = 1 + nR + mS + 1;
        int S = 0, T = nodes - 1;
        g = new List[nodes];
        for (int i = 0; i < nodes; i++) g[i] = new ArrayList<>();
        for (int i = 0; i < nR; i++) addEdge(S, 1 + i, 1);
        for (int j = 0; j < mS; j++) addEdge(1 + nR + j, T, scap[j]);
        for (int i = 0; i < nR; i++)
            for (int j = 0; j < mS; j++)
                if (Math.abs(rpos[i] - spos[j]) <= E) addEdge(1 + i, 1 + nR + j, 1);
        level = new int[nodes]; iter = new int[nodes];
        int flow = 0;
        while (bfs(S, T)) {
            Arrays.fill(iter, 0);
            int f;
            while ((f = dfs(S, T, Integer.MAX_VALUE)) > 0) {
                flow += f;
                if (flow >= nR) return true;
            }
        }
        return flow == nR;
    }

    public static void main(String[] args) throws IOException {
        StreamTokenizer in = new StreamTokenizer(new BufferedReader(new InputStreamReader(System.in)));
        in.nextToken(); int T = (int) in.nval;
        StringBuilder out = new StringBuilder();
        for (int tc = 0; tc < T; tc++) {
            in.nextToken(); nR = (int) in.nval;
            in.nextToken(); mS = (int) in.nval;
            rpos = new long[nR]; spos = new long[mS]; scap = new int[mS];
            for (int i = 0; i < nR; i++) { in.nextToken(); rpos[i] = (long) in.nval; }
            for (int j = 0; j < mS; j++) { in.nextToken(); spos[j] = (long) in.nval; }
            for (int j = 0; j < mS; j++) { in.nextToken(); scap[j] = (int) in.nval; }
            long totalCap = 0;
            for (int c : scap) totalCap += c;
            if (totalCap < nR) { out.append(-1).append('\n'); continue; }
            TreeSet<Long> set = new TreeSet<>();
            set.add(0L);
            for (int i = 0; i < nR; i++)
                for (int j = 0; j < mS; j++) set.add(Math.abs(rpos[i] - spos[j]));
            ArrayList<Long> list = new ArrayList<>(set);
            int lo = 0, hi = list.size() - 1, ansIdx = -1;
            while (lo <= hi) {
                int mid = (lo + hi) >>> 1;
                if (feasible(list.get(mid))) { ansIdx = mid; hi = mid - 1; }
                else lo = mid + 1;
            }
            out.append(ansIdx < 0 ? -1 : list.get(ansIdx)).append('\n');
        }
        System.out.print(out);
    }
}
```


### Python 实现

```python
# 题目：第2题
# 算法：BFS / 队列
# 模式：ACM（stdin/stdout）

import sys
from collections import deque

sys.setrecursionlimit(1000000)

# 核心函数：main
def main():
    data = sys.stdin.buffer.read().split()
    # 条件判断
    if not data:
        # 返回结果
        return
    it = iter(data)
    T = int(next(it))
    out = []
    # 遍历 _
    for _ in range(T):
        nR = int(next(it))
        mS = int(next(it))
        rpos = [int(next(it)) for _i in range(nR)]
        spos = [int(next(it)) for _i in range(mS)]
        scap = [int(next(it)) for _i in range(mS)]
        total_cap = 0
        # 遍历 c
        for c in scap:
            total_cap += c
        # 条件判断
        if total_cap < nR:
            out.append('-1')
            continue

        # 核心函数：feasible
        def feasible(E):
            nodes = 1 + nR + mS + 1
            S = 0
            Tk = nodes - 1
            g = [[] for _ in range(nodes)]

            # 核心函数：add_edge
            def add_edge(f, t, c):
                g[f].append([t, len(g[t]), c])
                g[t].append([f, len(g[f]) - 1, 0])

            # 枚举位置 i
            for i in range(nR):
                add_edge(S, 1 + i, 1)
            # 枚举阶段 j
            for j in range(mS):
                add_edge(1 + nR + j, Tk, scap[j])
            # 枚举位置 i
            for i in range(nR):
                # 枚举阶段 j
                for j in range(mS):
                    # 条件判断
                    if abs(rpos[i] - spos[j]) <= E:
                        add_edge(1 + i, 1 + nR + j, 1)
            level = [-1] * nodes
            iter_ = [0] * nodes

            # 核心函数：bfs
            def bfs():
                # 枚举位置 i
                for i in range(nodes):
                    level[i] = -1
                q = deque([S])
                level[S] = 0
                # 循环处理
                while q:
                    v = q.popleft()
                    # 遍历 e
                    for e in g[v]:
                        # 条件判断
                        if e[2] > 0 and level[e[0]] < 0:
                            level[e[0]] = level[v] + 1
                            q.append(e[0])
                # 返回结果
                return level[Tk] >= 0

            # 核心函数：dfs
            def dfs(v, f):
                # 条件判断
                if v == Tk:
                    # 返回结果
                    return f
                # 循环处理
                while iter_[v] < len(g[v]):
                    e = g[v][iter_[v]]
                    # 条件判断
                    if e[2] > 0 and level[v] + 1 == level[e[0]]:
                        d = dfs(e[0], min(f, e[2]))
                        # 条件判断
                        if d > 0:
                            e[2] -= d
                            g[e[0]][e[1]][2] += d
                            # 返回结果
                            return d
                    iter_[v] += 1
                # 返回结果
                return 0

            flow = 0
            # 循环处理
            while bfs():
                # 枚举位置 i
                for i in range(nodes):
                    iter_[i] = 0
                # 循环处理
                while True:
                    f = dfs(S, 1 << 60)
                    # 条件判断
                    if f <= 0:
                        break
                    flow += f
                    # 条件判断
                    if flow >= nR:
                        # 返回结果
                        return True
            # 返回结果
            return flow == nR

        cand = set()
        cand.add(0)
        # 枚举位置 i
        for i in range(nR):
            # 枚举阶段 j
            for j in range(mS):
                cand.add(abs(rpos[i] - spos[j]))
        lst = sorted(cand)
        lo = 0
        hi = len(lst) - 1
        ans_idx = -1
        # 循环处理
        while lo <= hi:
            mid = (lo + hi) // 2
            # 条件判断
            if feasible(lst[mid]):
                ans_idx = mid
                hi = mid - 1
            else:
                lo = mid + 1
        out.append(str(lst[ans_idx]) if ans_idx >= 0 else '-1')
    # 遍历 v
    for v in out:
        print(v)

main()
```

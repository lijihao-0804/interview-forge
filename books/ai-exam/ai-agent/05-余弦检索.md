# 余弦检索

> 来源：外部面经参考/02-AI与Agent手撕代码题.md · AI/Agent 手撕题

### Python 实现

```python
# 题目：余弦检索
# 算法：按题意模拟
# 模式：ACM（stdin/stdout）

import math


def cosine(left, right):
    numerator = sum(a * b for a, b in zip(left, right))
    left_norm = math.sqrt(sum(a * a for a in left))
    right_norm = math.sqrt(sum(b * b for b in right))
    if left_norm == 0 or right_norm == 0:
        return 0.0
    return numerator / (left_norm * right_norm)


def top_k(query, documents, k=3):
    scored = [(cosine(query, vector), doc_id) for doc_id, vector in documents]
    return sorted(scored, reverse=True)[:k]
```

### Java 实现

```java
// 题目：余弦检索
// 算法：排序 + 扫描
// 模式：ACM（stdin/stdout）

import java.util.ArrayList;
import java.util.List;
import java.util.Map;

final class CosineRetrieval {
    static double cosine(List<Double> left, List<Double> right) {
        double numerator = 0.0;
        int n = Math.min(left.size(), right.size());
        for (int i = 0; i < n; i++) {
            numerator += left.get(i) * right.get(i);
        }
        double leftNorm = 0.0;
        for (double value : left) {
            leftNorm += value * value;
        }
        double rightNorm = 0.0;
        for (double value : right) {
            rightNorm += value * value;
        }
        leftNorm = Math.sqrt(leftNorm);
        rightNorm = Math.sqrt(rightNorm);
        if (leftNorm == 0.0 || rightNorm == 0.0) {
            return 0.0;
        }
        return numerator / (leftNorm * rightNorm);
    }

    static final class Scored {
        final double score;
        final String docId;

        Scored(double score, String docId) {
            this.score = score;
            this.docId = docId;
        }
    }

    static List<Scored> topK(List<Double> query, List<Map.Entry<String, List<Double>>> documents, int k) {
        List<Scored> scored = new ArrayList<>();
        for (Map.Entry<String, List<Double>> document : documents) {
            scored.add(new Scored(cosine(query, document.getValue()), document.getKey()));
        }
        scored.sort((p, q) -> {
            int cmp = Double.compare(p.score, q.score);
            if (cmp != 0) {
                return -cmp;
            }
            return q.docId.compareTo(p.docId);
        });
        return new ArrayList<>(scored.subList(0, Math.min(k, scored.size())));
    }

    static List<Scored> topK(List<Double> query, List<Map.Entry<String, List<Double>>> documents) {
        return topK(query, documents, 3);
    }

    // 主函数：读取输入并输出答案
    public static void main(String[] args) {
        List<Map.Entry<String, List<Double>>> documents = new ArrayList<>();
        documents.add(new java.util.AbstractMap.SimpleEntry<>("a", java.util.Arrays.asList(1.0, 0.0)));
        documents.add(new java.util.AbstractMap.SimpleEntry<>("b", java.util.Arrays.asList(0.0, 1.0)));
        List<Scored> top = topK(java.util.Arrays.asList(1.0, 0.1), documents, 3);
        for (Scored scored : top) {
            System.out.println(scored.docId + " " + scored.score);
        }
    }
}
```

416. 分割等和子集

难度中等核心模式0-1 背包时间复杂度O(n·target)空间复杂度O(target)

题目与约束

给你一个 只包含正整数的 非空数组 nums 。请你判断是否可以将这个数组分割成两个子集，使得两个子集的元素和相等。

示例 1：

输入：nums = [1,5,11,5]
输出：true
解释：数组可以分割成 [1, 5, 5] 和 [11] 。

示例 2：

输入：nums = [1,2,3,5]
输出：false
解释：数组不能分割成两个元素和相等的子集。

提示：

  • 1 <= nums.length <= 200
  • 1 <= nums[i] <= 100

核心不变量

容量倒序保证每个数字只使用一次。

完整推导

思路推导

  1. 直觉与痛点（回溯的灾难）：

    最直观的想法是对每一个数字做抉择：“放入左集合”或“放入右集合”。这就是标准的回溯法（DFS）。

    痛点：每一个数字都有 2 种选择，时间复杂度高达 O(2N)。如果数组长度超过 30，直接超时到天荒地老。

  2. 降维破局（数学转换 + 背包登场）：

    怎么才能让两个集合的和相等？

    显然，整个数组的总和必须是偶数！如果总和是奇数，比如 11，你怎么可能把它平分成两个整数？直接判死刑返回 false。

    如果总和是偶数（假设是 22），那么目标就极其明确了：我们只需要从数组里挑出几个数，让它们的和恰好等于总和的一半（11）即可。

    一旦挑出了和为 11 的子集，剩下的数字自然而然和也是 11。

    这完美的契合了“0-1 背包”的模型：有一个容量为 11 的背包，每个数字是一件物品（重量和价值都是数字本身），每件物品只能用一次，问能不能刚好把背包塞满？

  3. 核心微操（状态转移方程）：

    我们定义一个布尔数组 dp[j]：表示能否从数组中挑出一些数字，使得它们的和恰好等于 j。

    假设我们正准备考虑数字 nums[i]，要更新 dp[j] 时，我们有两种选择：

    • 不把 nums[i] 放入背包：那能否凑出 j，完全取决于历史记录 dp[j] 本身是不是 true。

    • 把 nums[i] 放入背包：那能否凑出 j，取决于我们在没放它之前，能否凑出 j - nums[i]。也就是看历史记录 dp[j - nums[i]] 是不是 true。

    只要这两种情况有一种为 true，现在的 dp[j] 就是 true。

    方程：dp[j]=dp[j]||dp[j-nums[i]]

    Java 实现（满分空间压缩版）

    C++ 实现

    实现来源：灵茶山艾府题解（MIT License）

    class Solution {
public:
    bool canPartition(vector<int>& nums) {
        int s = reduce(nums.begin(), nums.end());
        if (s % 2) {
            return false;
        }

        int n = nums.size();
        vector memo(n, vector<int>(s / 2 + 1, -1)); // -1 表示没有计算过

        // lambda 递归函数
        auto dfs = [&](this auto&& dfs, int i, int j) -> bool {
            if (i < 0) {
                return j == 0;
            }

            int& res = memo[i][j]; // 注意这里是引用
            if (res != -1) { // 之前计算过
                return res;
            }

            if (j < nums[i]) {
                return res = dfs(i - 1, j); // 只能不选
            }
            return res = dfs(i - 1, j - nums[i]) || dfs(i - 1, j); // 选或不选
        };

        return dfs(n - 1, s / 2);
    }
};

    Python 实现

    实现来源：灵茶山艾府题解（MIT License）

    class Solution:
    def canPartition(self, nums: List[int]) -> bool:
        _sum = 0

        # dp[i]中的i表示背包内总和
        # 题目中说：每个数组中的元素不会超过 100，数组的大小不会超过 200
        # 总和不会大于20000，背包最大只需要其中一半，所以10001大小就可以了
        dp = [0] * 10001
        for num in nums:
            _sum += num
        # 也可以使用内置函数一步求和
        # _sum = sum(nums)
        if _sum % 2 == 1:
            return False
        target = _sum // 2

        # 开始 0-1背包
        for num in nums:
            for j in range(target, num - 1, -1):  # 每一个元素一定是不可重复放入，所以从大到小遍历
                dp[j] = max(dp[j], dp[j - num] + num)

        # 集合中的元素正好可以凑成总和target
        if dp[target] == target:
            return True
        return False

    Go 实现

    实现来源：灵茶山艾府题解（MIT License）

    func canPartition(nums []int) bool {
    s := 0
    for _, x := range nums {
        s += x
    }
    if s%2 != 0 {
        return false
    }

    n := len(nums)
    memo := make([][]int8, n)
    for i := range memo {
        memo[i] = make([]int8, s/2+1)
        for j := range memo[i] {
            memo[i][j] = -1 // -1 表示没有计算过
        }
    }

    var dfs func(int, int) bool
    dfs = func(i, j int) (res bool) {
        if i < 0 {
            return j == 0
        }
        p := &memo[i][j]
        if *p != -1 { // 之前计算过
            return *p == 1
        }

        if j < nums[i] {
            res = dfs(i-1, j) // 只能不选
        } else {
            res = dfs(i-1, j-nums[i]) || dfs(i-1, j) // 选或不选
        }
        // 记忆化
        if res {
            *p = 1
        } else {
            *p = 0
        }
        return
    }

    return dfs(n-1, s/2)
}

    C 实现

    实现来源：灵茶山艾府题解（MIT License）

    bool canPartition(int* nums, int numsSize) {
    int s = 0;
    for (int i = 0; i < numsSize; i++) {
        s += nums[i];
    }
    if (s % 2) {
        return false;
    }

    int** memo = malloc(numsSize * sizeof(int*));
    for (int i = 0; i < numsSize; i++) {
        memo[i] = malloc((s / 2 + 1) * sizeof(int));
        memset(memo[i], -1, (s / 2 + 1) * sizeof(int)); // -1 表示没有计算过
    }

    bool dfs(int i, int j) {
        if (i < 0) {
            return j == 0;
        }
        if (memo[i][j] != -1) { // 之前计算过
            return memo[i][j] == 1;
        }

        if (j < nums[i]) {
            return memo[i][j] = dfs(i - 1, j); // 只能不选
        }
        return memo[i][j] = dfs(i - 1, j - nums[i]) || dfs(i - 1, j); // 选或不选
    }

    int ans = dfs(numsSize - 1, s / 2);
    for (int i = 0; i < numsSize; i++) {
        free(memo[i]);
    }
    free(memo);
    return ans;
}

class Solution {
    public boolean canPartition(int[] nums) {
        int sum = 0;
        for (int num : nums) {
            sum += num;
        }

        // 1. 致命剪枝：如果总和是奇数，绝对不可能平分
        if (sum % 2 != 0) {
            return false;
        }

        int target = sum / 2;

        // dp[j] 表示能否凑出和为 j 的子集
        boolean[] dp = new boolean[target + 1];

        // 🌟 初始化基石：凑出和为 0 是永远可以的（什么都不选就行）
        dp[0] = true;

        // 外层循环：遍历每一件“物品”（每个数字）
        for (int i = 0; i < nums.length; i++) {
            // 🌟 致命工程细节：内层循环【必须倒序遍历】！
            // 从 target 一直往回退，退到装不下当前物品 nums[i] 为止
            for (int j = target; j >= nums[i]; j--) {
                // 状态转移：要么不选当前物品（维持 dp[j]），要么选当前物品（依赖 dp[j - nums[i]]）
                dp[j] = dp[j] || dp[j - nums[i]];
            }

            // 性能优化（剪枝）：如果在遍历中途，dp[target] 已经变成 true 了，直接提前胜利下班！
            if (dp[target]) {
                return true;
            }
        }

        return dp[target];
    }
}

复杂度

  • 时间复杂度：O(N×target)。其中 N 是数组长度，target 是总和的一半。典型的伪多项式时间复杂度。
  • 空间复杂度：O(target)。仅仅开辟了一个长度为 target + 1 的一维布尔数组，极致的空间压缩。
易错点与扩展（面试的关键拷问）
  • 面试面试官最爱抓的破绽：“在『完全背包』（如零钱兑换）里，内层循环是从小到大正序遍历的；为什么在这道『0-1 背包』里，内层循环【必须从大到小倒序遍历】？”
  • 满分保命回答（时空倒流法则）：
    • 如果我们正序遍历：假设当前物品是 2。我们在算 dp[2] 时用到了 dp[0] 变成了 true。紧接着算 dp[4] 时，会去查 dp[2]。此时的 dp[2] 刚刚被这件物品 2 染成了 true，导致 dp[4] 也变成了 true！这等于把同一件物品 2 用了两次！这就退化成了“完全背包”问题。
    • 如果我们倒序遍历：算 dp[4] 时去查 dp[2]，此时由于还没遍历到索引 2，所以这个 dp[2] 绝对是上一轮历史留下来的干净数据（没有被当前物品污染过）。倒序遍历完美保证了：在更新当前层的 dp 时，依赖的永远是上一层的状态，从而确保每件物品严格只被使用一次！

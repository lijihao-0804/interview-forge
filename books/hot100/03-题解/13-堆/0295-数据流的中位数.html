295. 数据流的中位数

← 堆专题 · 总目录 · 复习清单

难度困难核心模式大顶堆 + 小顶堆时间复杂度插入 O(log n)，查询 O(1)空间复杂度O(n)

题目与约束

中位数是有序整数列表中的中间值。如果列表的大小是偶数，则没有中间值，中位数是两个中间值的平均值。

  • 例如 arr = [2,3,4] 的中位数是 3 。
  • 例如 arr = [2,3] 的中位数是 (2 + 3) / 2 = 2.5 。

实现 MedianFinder 类:

  • MedianFinder() 初始化 MedianFinder 对象。
  • void addNum(int num) 将数据流中的整数 num 添加到数据结构中。
  • double findMedian() 返回到目前为止所有元素的中位数。与实际答案相差 10-5 以内的答案将被接受。

示例 1：

输入
["MedianFinder", "addNum", "addNum", "findMedian", "addNum", "findMedian"]
[[], [1], [2], [], [3], []]
输出
[null, null, null, 1.5, null, 2.0]

解释
MedianFinder medianFinder = new MedianFinder();
medianFinder.addNum(1); // arr = [1]
medianFinder.addNum(2); // arr = [1, 2]
medianFinder.findMedian(); // 返回 1.5 ((1 + 2) / 2)
medianFinder.addNum(3); // arr[1, 2, 3]
medianFinder.findMedian(); // return 2.0

提示:

  • -105 <= num <= 105
  • 在调用 findMedian 之前，数据结构中至少有一个元素

  • 最多 5 * 104 次调用 addNum 和 findMedian

  • 时间复杂度：

  • addNum：O(log N)。每次插入和弹出操作都是对二叉堆的调整，代价是对数级别的。
  • findMedian：O(1)。中位数永远就在两个堆的嘴尖上（堆顶），看一眼直接返回，极速响应。
  • 空间复杂度：O(N)。需要两个堆合力装下数据流里的所有数字。

核心不变量

两堆元素数之差不超过一，且左堆所有值不大于右堆。

完整推导

思路推导

  1. 直觉与痛点（动态排序的绝望）：

中位数把所有数字分成了“左边较小的一半”和“右边较大的一半”。

如果用动态数组 ArrayList 维护，新来一个数，为了保持有序，我们需要用二分查找找到位置并插入，移动元素需要 O(N) 的时间。数据量一旦大起来，动态插入会彻底卡死。

  1. 降维破局（双堆对顶，各据一方）：

我们其实根本不需要让所有数字都严格有序！我们只需要牢牢把控住“左半区的最大值”和“右半区的最小值”。

由此，我们创造出两个大本营：

  • 左半区：大顶堆（maxHeap），保存较小的一半，堆顶是左半区最大值。
  • 右半区：小顶堆（minHeap），保存较大的一半，堆顶是右半区最小值。
  • 插入时先放进左半区，再把左半区最大值移到右半区，最后平衡两堆大小，保证左堆数量等于右堆数量，或左堆比右堆多 1。
  • 取中位数时，两堆数量相等就取两个堆顶的平均值，否则取左堆堆顶。

Java 实现

import java.util.Comparator;
import java.util.PriorityQueue;

class MedianFinder {
    // 左半区：大顶堆，保存较小的一半，堆顶是左半区最大值
    private PriorityQueue<Integer> maxHeap;
    // 右半区：小顶堆，保存较大的一半，堆顶是右半区最小值
    private PriorityQueue<Integer> minHeap;

    public MedianFinder() {
        maxHeap = new PriorityQueue<>(Comparator.reverseOrder());
        minHeap = new PriorityQueue<>();
    }

    public void addNum(int num) {
        // 先放进左半区，再把左半区最大值移到右半区，保证左堆所有值 <= 右堆所有值
        maxHeap.offer(num);
        minHeap.offer(maxHeap.poll());
        // 平衡两堆大小：左堆数量要么等于右堆，要么比右堆多 1
        if (minHeap.size() > maxHeap.size()) {
            maxHeap.offer(minHeap.poll());
        }
    }

    public double findMedian() {
        // 奇数个元素时，中位数就是左堆堆顶
        if (maxHeap.size() > minHeap.size()) {
            return maxHeap.peek();
        }
        // 偶数个元素时，取两个堆顶的平均值
        return (maxHeap.peek() + minHeap.peek()) / 2.0;
    }
}

C++ 实现

实现来源：经典解法实现

  1. 双堆：大根堆存较小的一半，小根堆存较大的一半，两堆大小差 ≤1
  2. addNum：先按值入对应堆，再平衡两堆大小
  3. findMedian：奇数个取大堆顶；偶数个取两堆顶平均
class MedianFinder {
    priority_queue<int> lo;                        // 大根堆：较小一半
    priority_queue<int, vector<int>, greater<int>> hi;  // 小根堆：较大一半
public:
    void addNum(int num) {
        lo.push(num);
        hi.push(lo.top()); lo.pop();
        if (hi.size() > lo.size()) { lo.push(hi.top()); hi.pop(); }
    }
    double findMedian() {
        return lo.size() > hi.size() ? lo.top() : (lo.top() + hi.top()) / 2.0;
    }
};

Python 实现

实现来源：经典解法实现

  1. 双堆：大根堆存较小的一半，小根堆存较大的一半，两堆大小差 ≤1
  2. addNum：先按值入对应堆，再平衡两堆大小
  3. findMedian：奇数个取大堆顶；偶数个取两堆顶平均
import heapq
class MedianFinder:
    def __init__(self):
        self.lo, self.hi = [], []       # lo 大根堆（负数技巧），hi 小根堆
    def addNum(self, num: int) -> None:
        heapq.heappush(self.lo, -num)
        heapq.heappush(self.hi, -heapq.heappop(self.lo))
        if len(self.hi) > len(self.lo):
            heapq.heappush(self.lo, -heapq.heappop(self.hi))
    def findMedian(self) -> float:
        return -self.lo[0] if len(self.lo) > len(self.hi) else (-self.lo[0] + self.hi[0]) / 2

Go 实现

实现来源：经典解法实现

  1. 双堆：大根堆存较小的一半，小根堆存较大的一半，两堆大小差 ≤1
  2. addNum：先按值入对应堆，再平衡两堆大小
  3. findMedian：奇数个取大堆顶；偶数个取两堆顶平均
type MedianFinder struct {
    lo, hi []int   // lo 大根堆（存负数），hi 小根堆
}
func Constructor() MedianFinder { return MedianFinder{} }
func (mf *MedianFinder) AddNum(num int) {
    heap.Push(&maxHeap{&mf.lo}, num)
    heap.Push(&minHeap{&mf.hi}, heap.Pop(&maxHeap{&mf.lo}))
    if len(mf.hi) > len(mf.lo) { heap.Push(&maxHeap{&mf.lo}, heap.Pop(&minHeap{&mf.hi})) }
}
func (mf *MedianFinder) FindMedian() float64 {
    if len(mf.lo) > len(mf.hi) { return float64(mf.lo[0]) }
    return float64(mf.lo[0]-mf.hi[0]) / 2 * -1
}

C 实现

实现来源：经典解法实现

  1. 双堆：大根堆存较小的一半，小根堆存较大的一半，两堆大小差 ≤1
  2. addNum：先按值入对应堆，再平衡两堆大小
  3. findMedian：奇数个取大堆顶；偶数个取两堆顶平均
typedef struct { int* lo; int* hi; int loN, hiN, cap; } MedianFinder;
void upLo(int* a, int i) { while (i && a[(i-1)/2] < a[i]) { int t = a[i]; a[i] = a[(i-1)/2]; a[(i-1)/2] = t; i = (i-1)/2; } }
void downLo(int* a, int n, int i) { while (1) { int l = 2*i+1, r = 2*i+2, big = i; if (l < n && a[l] > a[big]) big = l; if (r < n && a[r] > a[big]) big = r; if (big == i) break; int t = a[i]; a[i] = a[big]; a[big] = t; i = big; } }
void upHi(int* a, int i) { while (i && a[(i-1)/2] > a[i]) { int t = a[i]; a[i] = a[(i-1)/2]; a[(i-1)/2] = t; i = (i-1)/2; } }
downHi(int* a, int n, int i) { while (1) { int l = 2*i+1, r = 2*i+2, sm = i; if (l < n && a[l] < a[sm]) sm = l; if (r < n && a[r] < a[sm]) sm = r; if (sm == i) break; int t = a[i]; a[i] = a[sm]; a[sm] = t; i = sm; } }
MedianFinder* medianFinderCreate() { MedianFinder* mf = malloc(sizeof(MedianFinder)); mf->cap = 64; mf->loN = mf->hiN = 0; mf->lo = malloc(4 * mf->cap); mf->hi = malloc(4 * mf->cap); return mf; }
void medianFinderAddNum(MedianFinder* mf, int num) {
    mf->lo[mf->loN] = num; upLo(mf->lo, mf->loN); mf->loN++;
    mf->hi[mf->hiN] = mf->lo[0]; mf->hiN++; mf->lo[0] = mf->lo[--mf->loN]; downLo(mf->lo, mf->loN, 0);
    if (mf->hiN > mf->loN) { mf->lo[mf->loN] = mf->hi[0]; upLo(mf->lo, mf->loN); mf->loN++; mf->hi[0] = mf->hi[--mf->hiN]; downHi(mf->hi, mf->hiN, 0); }
}
double medianFinderFindMedian(MedianFinder* mf) {
    return mf->loN > mf->hiN ? mf->lo[0] : (mf->lo[0] + mf->hi[0]) / 2.0;
}
易错点与扩展（面试高频连环追问）
  • 面试进阶追问 1：“如果数据流中 99% 的数字都在 0 到 100 之间，你还用双堆吗？”
  • 满分抢答（计数排序大显神威）：这时候用堆就亏大了！既然数字范围极度紧凑（[0, 100]），我们应该直接开辟一个长度为 101 的计数数组（Bucket）！
  • 每次来一个数，对应的频次加 1（count[num]++），耗时 O(1)。
  • 寻找中位数时，我们维护一个全局总数 N，直接从 0 到 100 累加频次，找到第 N/2 个数字对应的位置。因为桶的长度只有 101，这个遍历也是 O(1) 的常数级！空间更是压缩到了极点。
  • 面试进阶追问 2：“如果 99% 的数字在 0 到 100 之间，但有 1% 的数字是不可预测的巨大超级大离群点（Outliers）呢？”
  • 高级风控架构方案：我们可以把两个方案缝合起来！
  • 用一个长度为 101 的数组记录 [0, 100] 内的正常数字，并记录低于 0 的个数和高于 100 的个数。
  • 对于那些极少数越界的超级大离群点，我们单独用两个小规模的双堆去挂载在两头。在寻找中位数时，先排除两边的离群点数量，再定位到中央的计数数组中。这种“主仓+边缘流控”的设计思想，直接对齐现代分布式大数据的风控模型。

for (int i = 0; i < nums.length; i++) {

    int size = queue.size();
    for (int i = 0; i < size; i++) {

9. 网格 DFS

void dfs(char[][] grid, int row, int col) {
    if (row < 0 || row >= grid.length || col < 0 || col >= grid[0].length
            || grid[row][col] != '1') return;
    grid[row][col] = '0'; // 进入即标记
    dfs(grid, row + 1, col);
    dfs(grid, row - 1, col);
    dfs(grid, row, col + 1);
    dfs(grid, row, col - 1);
}

Python 算法模板

下列 17 段模板各自对应一个可执行的代表例子。模板用于对照变量如何移动；换题时要按题意重新定义键、区间和状态，不能只替换函数名。

哈希表：先查后存的补数表

哈希表把已经走过的元素加工成可快速查询的键。先决定当前元素要查什么，再决定键与值。

参考来源：代码随想录（programmercarl.com）

这是“两数之和”的可运行最小例子。seen 的键是数值，值是下标；这一写法依赖题目的一遍补数关系，换成分组、计数或前缀和题时，键值含义需要重写。

def two_sum(nums: list[int], target: int) -> list[int]:
    seen = {}
    for index, value in enumerate(nums):
        need = target - value
        if need in seen:
            return [seen[need], index]
        seen[value] = index
    return []

双指针：相向指针排除短板

双指针用一个区间或两个速度表达可证明的排除关系；两侧指针各有明确任务，不能只凭写法套模板。

参考来源：代码随想录（programmercarl.com）

模板解决“盛最多水的容器”。相向移动适用于能从两端单调性排除答案的题；移动零一类的同向覆盖要另定义 write 指针，不能照搬这里的移动条件。

def max_area(height: list[int]) -> int:
    left = 0
    right = len(height) - 1
    best = 0
    while left < right:
        width = right - left
        area = min(height[left], height[right]) * width
        best = max(best, area)
        if height[left] <= height[right]:
            left += 1
        else:
            right -= 1
    return best

滑动窗口：最近位置窗口

滑动窗口维护一段连续区间；右端加入新元素，左端只在不变量被破坏或需要缩短时移动。

参考来源：代码随想录（programmercarl.com）

模板解决“无重复字符的最长子串”。它适合用最近位置一次跳转左端；438 题是固定长度频次窗口，76 题是最小覆盖窗口，移除步骤和答案更新时机都不同。

def length_of_longest_substring(s: str) -> int:
    last = {}
    left = 0
    best = 0
    for right, ch in enumerate(s):
        if ch in last:
            left = max(left, last[ch] + 1)
        last[ch] = right
        best = max(best, right - left + 1)
    return best

子串：前缀和频次

连续区间的“和”“最值”“覆盖”需要不同工具：区间和查前缀，定长最值用单调队列，覆盖条件用可变窗口。

参考来源：代码随想录（programmercarl.com）

模板解决元素可为负数的“和为 K 的子数组”。遇到固定长度窗口最值或覆盖窗口时，应改看 deque 或频次表模板；前缀和不能直接替换窗口最值。

def subarray_sum(nums: list[int], k: int) -> int:
    seen = {0: 1}
    prefix = 0
    answer = 0
    for value in nums:
        prefix += value
        answer += seen.get(prefix - k, 0)
        seen[prefix] = seen.get(prefix, 0) + 1
    return answer

普通数组：Kadane 滚动状态

数组题先明确下标含义，再决定原地扫描、排序、前后缀或值域映射；不是看到数组就先切片。

参考来源：代码随想录（programmercarl.com）

模板解决非空“最大子数组和”。current 必须定义为“以当前位置结尾”的最大值；区间合并、三次反转和值域置换是另外三类数组模型。

def kadane(nums: list[int]) -> int:
    current = nums[0]
    best = nums[0]
    for index in range(1, len(nums)):
        value = nums[index]
        current = max(value, current + value)
        best = max(best, current)
    return best

矩阵：四边界螺旋模拟

矩阵题先固定 rows、cols 与边界开闭语义，再决定螺旋模拟、首行首列标记、转置翻转或阶梯搜索。

参考来源：代码随想录（programmercarl.com）

模板解决非空或空列表都可安全处理的螺旋遍历。每收缩一条边都要重新检查剩余区域；置零、旋转和二维搜索需要各自的状态，不能共用这一个模板。

def spiral(matrix: list[list[int]]) -> list[int]:
    if not matrix or not matrix[0]:
        return []
    rows = len(matrix)
    cols = len(matrix[0])
    top = 0
    bottom = rows - 1
    left = 0
    right = cols - 1
    answer = []
    while top <= bottom and left <= right:
        for col in range(left, right + 1):
            answer.append(matrix[top][col])
        top += 1
        for row in range(top, bottom + 1):
            answer.append(matrix[row][right])
        right -= 1
        if top <= bottom:
            for col in range(right, left - 1, -1):
                answer.append(matrix[bottom][col])
            bottom -= 1
        if left <= right:
            for row in range(bottom, top - 1, -1):
                answer.append(matrix[row][left])
            left += 1
    return answer

链表：四步原地反转

链表题先画前驱、当前与后继。修改 next 前保存后路，复杂重连优先用虚拟头节点固定入口。

参考来源：代码随想录（programmercarl.com）

模板解决 206 反转链表；ListNode 由力扣判题入口提供，本地运行时可传入任意具有 next 属性的节点对象。它原地改链，调用后原头会变成尾。

from typing import Optional


def reverse_list(head: Optional["ListNode"]) -> Optional["ListNode"]:
    previous = None
    current = head
    while current is not None:
        following = current.next
        current.next = previous
        previous = current
        current = following
    return previous

二叉树：后序返回树高

树题先确定信息流向：父节点向下传边界，子节点向上返回高度或贡献，BFS 按层处理，BST 利用中序。

参考来源：代码随想录（programmercarl.com）；原始递归思路，本模板改为显式栈

模板返回二叉树高度，空树返回 0。它用 (node, expanded) 显式表示后序的进入与返回，TreeNode 由力扣提供；本地测试时传入具有 left、right 属性的节点即可。

from __future__ import annotations


def tree_height(root: TreeNode | None) -> int:
    if root is None:
        return 0
    heights = {}
    stack = [(root, False)]
    while stack:
        node, expanded = stack.pop()
        if expanded:
            left_height = 0 if node.left is None else heights[node.left]
            right_height = 0 if node.right is None else heights[node.right]
            heights[node] = 1 + max(left_height, right_height)
            continue
        stack.append((node, True))
        if node.right is not None:
            stack.append((node.right, False))
        if node.left is not None:
            stack.append((node.left, False))
    return heights[root]

图论：入队即标记的岛屿洪泛

网格是隐式图，显式依赖先建邻接表。搜索状态必须在进入容器时标记，多源问题先把所有源同时入队。

参考来源：代码随想录（programmercarl.com）

模板解决 200 题，并把访问过的陆地原地改成 "0"。若调用方之后还需要原网格，应先复制；多源最短路径和拓扑排序要另建状态，不能只换函数名。

from collections import deque


def count_islands(grid: list[list[str]]) -> int:
    if not grid or not grid[0]:
        return 0
    rows = len(grid)
    cols = len(grid[0])
    islands = 0
    directions = ((1, 0), (-1, 0), (0, 1), (0, -1))
    for start_row in range(rows):
        for start_col in range(cols):
            if grid[start_row][start_col] != "1":
                continue
            islands += 1
            grid[start_row][start_col] = "0"
            queue = deque([(start_row, start_col)])
            while queue:
                row, col = queue.popleft()
                for dr, dc in directions:
                    next_row = row + dr
                    next_col = col + dc
                    if not (0 <= next_row < rows and 0 <= next_col < cols):
                        continue
                    if grid[next_row][next_col] != "1":
                        continue
                    grid[next_row][next_col] = "0"
                    queue.append((next_row, next_col))
    return islands

回溯：起点索引子集树

回溯把题目写成一棵决策树：当前路径、下一层候选、终止条件与撤销动作共同决定搜索结果。

参考来源：代码随想录（programmercarl.com）

模板解决元素互不相同的 78 题子集。start 保证只向后选，避免同一种子集换顺序重复；全排列、可重复组合和棋盘回溯需要各自的选择与剪枝条件。

def subsets(nums: list[int]) -> list[list[int]]:
    answer = []
    path = []

    def backtrack(start: int) -> None:
        answer.append(path[:])
        for index in range(start, len(nums)):
            path.append(nums[index])
            backtrack(index + 1)
            path.pop()

    backtrack(0)
    return answer

二分查找：右开区间下界

二分查找先固定区间的开闭语义，再用单调条件排除一半候选；每一轮都必须严格缩小。

参考来源：代码随想录（programmercarl.com）

模板返回排序数组 nums 中第一个大于等于 target 的下标；没有任何元素满足时返回 len(nums)。右开区间模板不能随意改成 left <= right。

def lower_bound(nums: list[int], target: int) -> int:
    left = 0
    right = len(nums)
    while left < right:
        middle = (left + right) // 2
        if nums[middle] < target:
            left = middle + 1
        else:
            right = middle
    return left

栈：等待室单调下标栈

栈保存尚未闭合的结构，或保存仍在等待答案的元素；先说明栈中存值还是下标，再说明何时弹出。

参考来源：代码随想录（programmercarl.com）

模板解决 739 题。栈保存还没有等到更高温度的日期下标；相同温度仍会留在栈中，所以栈内温度是非递增而不是严格递减。

def daily_temperatures(temperatures: list[int]) -> list[int]:
    answer = [0] * len(temperatures)
    stack = []
    for today, temperature in enumerate(temperatures):
        while stack and temperature > temperatures[stack[-1]]:
            previous = stack.pop()
            answer[previous] = today - previous
        stack.append(today)
    return answer

堆：容量为 k 的频次小顶堆

堆用于持续取得候选中的极值。先决定堆顶应是什么，再决定何时把不够好的元素挤出去。

参考来源：代码随想录（programmercarl.com）

模板解决 347 题的前 k 个高频元素。堆按 (频次, 数值) 比较并只保留 k 项；题目未规定同频元素的先后时可返回其中任意一组。

from collections import Counter
import heapq


def top_k_frequent(nums: list[int], k: int) -> list[int]:
    counts = Counter(nums)
    heap = []
    for value, frequency in counts.items():
        heapq.heappush(heap, (frequency, value))
        if len(heap) > k:
            heapq.heappop(heap)
    answer = [0] * k
    for index in range(k - 1, -1, -1):
        answer[index] = heapq.heappop(heap)[1]
    return answer

贪心：历史最低价滚动

贪心每一步提交局部选择，并由交换或反证说明被放弃的选择不会让未来更优。

参考来源：代码随想录（programmercarl.com）

模板解决 121 题：只能先买后卖且交易一次。lowest 包含当天价格，但当天卖出利润为 0；跳跃游戏和字符分段使用不同的最远边界，不能直接套用。

def max_profit(prices: list[int]) -> int:
    lowest = float("inf")
    best = 0
    for price in prices:
        lowest = min(lowest, price)
        best = max(best, price - lowest)
    return best

动态规划：一维 0-1 背包可达性

动态规划先写状态含义，再写转移、初始化与遍历方向；空间压缩必须保留转移真正需要的旧状态。

参考来源：灵茶山艾府题解（MIT License）

模板解决 416 题：每个元素最多使用一次，目标是恰好凑出数组总和的一半。完全背包、序列 DP 和区间 DP 的遍历方向不同，需要重新推导。 300 题逐题抓手采用以每个位置结尾的 O(n²) DP，tails + bisect_left 只作扩展。

def can_partition(nums: list[int]) -> bool:
    total = sum(nums)
    if total % 2:
        return False
    target = total // 2
    reachable = [False] * (target + 1)
    reachable[0] = True
    for value in nums:
        for capacity in range(target, value - 1, -1):
            reachable[capacity] = reachable[capacity] or reachable[capacity - value]
    return reachable[target]

多维动态规划：网格最小路径和

二维 DP 的每个坐标都要有一句完整定义；先填边界，再确认当前格只依赖已经完成的相邻状态。

参考来源：经典解法实现

模板解决非空网格的 64 题，保留完整二维表以便逐步核对。它没有修改输入；若使用一维滚动，必须把“上方旧值”与“左方新值”分别保存。

def min_path_sum(grid: list[list[int]]) -> int:
    rows = len(grid)
    cols = len(grid[0])
    dp = [[0] * cols for _ in range(rows)]
    dp[0][0] = grid[0][0]
    for col in range(1, cols):
        dp[0][col] = dp[0][col - 1] + grid[0][col]
    for row in range(1, rows):
        dp[row][0] = dp[row - 1][0] + grid[row][0]
    for row in range(1, rows):
        for col in range(1, cols):
            dp[row][col] = grid[row][col] + min(
                dp[row - 1][col], dp[row][col - 1]
            )
    return dp[rows - 1][cols - 1]

def sort_colors(nums: list[int]) -> None:
    left = 0
    index = 0
    right = len(nums) - 1
    while index <= right:
        if nums[index] == 0:
            nums[left], nums[index] = nums[index], nums[left]
            left += 1
            index += 1
        elif nums[index] == 2:
            nums[index], nums[right] = nums[right], nums[index]
            right -= 1
        else:
            index += 1

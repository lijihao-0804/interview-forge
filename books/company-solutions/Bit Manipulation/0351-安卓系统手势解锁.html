351. 安卓系统手势解锁

难度中等核心模式回溯 DFS + 「跨点」合法性表时间复杂度O(9!)（板面固定，实际上是常数级）空间复杂度O(9)

题目与约束

🔗 前往力扣原题验证 →

核心不变量

从 cur 走到 nxt 合法的充要条件是：二者连线的中点 skip[cur][nxt] == 0（不跨任何键），或该中点按键已经被访问过。

完整推导

思路推导

  1. 直觉与痛点（看起来是"9 个键任排"）

最朴素的想法是“从 9 个键里按顺序不断挑没选过的键”，长度在 [m, n] 之间。但规则 3（不能跳过未选键） 打破了单纯的排列：像 1 → 3 这种穿过中心 2 的一步，只有当 2 已被选时才合法。所以必须在搜索时额外判断每一步是否"跨点合法"。

  1. 预计算"跨点表" skip

3×3 里会让连线穿过第三点的只有几组固定对称关系，把它们预先存进 skip[a][b]，值就是被穿过的中点编号（没有跨点则记 0）：

  • 同行/同列隔一个：1-3 跨 2，1-7 跨 4，3-9 跨 6，7-9 跨 8；
  • 对角线或对边中点：1-9 跨 5，2-8 跨 5，3-7 跨 5，4-6 跨 5。

至于 1-2、2-3 这类相邻键，skip = 0，永远合法。

  1. 回溯搜索 + 合法性判断

dfs(cur, len) 表示当前停在 cur、已选 len 个键。枚举下一个未访问的 nxt：取 mid = skip[cur][nxt]，只有当 mid == 0 或 visited[mid] 时才允许递归。当 len >= m 时当前前缀本身就算一个合法手势，先 +1 再继续向下扩展；len > n 时剪枝返回。搜索完记得把 cur 的访问标记撤销（回溯）。

  1. 对称性剪枝（把 9 次起点压到 3 次）

键盘高度对称：1,3,7,9 四个角完全等价，2,4,6,8 四条边中点等价，5 自成一类。所以只需从 1、2、5 各算一次，分别乘以 4、4、1 即可，省掉约 2/3 的重复计算。

Java 实现（numberOfPatterns）

class Solution {
    private boolean[] visited = new boolean[10];
    private int[][] skip = new int[10][10];

    public int numberOfPatterns(int m, int n) {
        // 预计算跨点表：skip[a][b] = a、b 连线穿过的中点，无跨点为 0
        skip[1][3] = skip[3][1] = 2;
        skip[1][7] = skip[7][1] = 4;
        skip[3][9] = skip[9][3] = 6;
        skip[7][9] = skip[9][7] = 8;
        skip[1][9] = skip[9][1] = 5;
        skip[2][8] = skip[8][2] = 5;
        skip[3][7] = skip[7][3] = 5;
        skip[4][6] = skip[6][4] = 5;

        int res = 0;
        res += dfs(1, 1, m, n) * 4; // 1,3,7,9 四个角对称
        res += dfs(2, 1, m, n) * 4; // 2,4,6,8 四条边中点对称
        res += dfs(5, 1, m, n);     // 5 独立，不乘系数
        return res;
    }

    // cur: 当前所在键；len: 当前手势已选键数
    private int dfs(int cur, int len, int m, int n) {
        if (len > n) return 0;                 // 超过上界，剪枝
        int cnt = (len >= m) ? 1 : 0;          // 达到下界，当前前缀算一种
        visited[cur] = true;
        for (int nxt = 1; nxt <= 9; nxt++) {
            if (visited[nxt]) continue;        // 不能重复选键
            int mid = skip[cur][nxt];
            if (mid == 0 || visited[mid]) {    // 不跨点，或跨过的点已选
                cnt += dfs(nxt, len + 1, m, n);
            }
        }
        visited[cur] = false;                  // 回溯：撤销访问标记
        return cnt;
    }
}

C++ 实现

#include <cstring>

class Solution {
    int skip[10][10];
    bool visited[10];

public:
    int numberOfPatterns(int m, int n) {
        memset(skip, 0, sizeof(skip));
        memset(visited, 0, sizeof(visited));

        skip[1][3] = skip[3][1] = 2;
        skip[1][7] = skip[7][1] = 4;
        skip[3][9] = skip[9][3] = 6;
        skip[7][9] = skip[9][7] = 8;
        skip[1][9] = skip[9][1] = skip[2][8] = skip[8][2] = 5;
        skip[3][7] = skip[7][3] = skip[4][6] = skip[6][4] = 5;

        int res = 0;
        res += dfs(1, 1, m, n) * 4;  // 四角对称
        res += dfs(2, 1, m, n) * 4;  // 四边中点对称
        res += dfs(5, 1, m, n);      // 中心独立
        return res;
    }

private:
    int dfs(int cur, int len, int m, int n) {
        if (len > n) return 0;
        int cnt = (len >= m) ? 1 : 0;
        visited[cur] = true;
        for (int nxt = 1; nxt <= 9; ++nxt) {
            if (visited[nxt]) continue;
            int mid = skip[cur][nxt];
            if (mid == 0 || visited[mid]) {
                cnt += dfs(nxt, len + 1, m, n);
            }
        }
        visited[cur] = false;   // 回溯
        return cnt;
    }
};

Python 实现

class Solution:
    def numberOfPatterns(self, m: int, n: int) -> int:
        # skip[a][b] = a、b 连线穿过的中点，无跨点为 0
        skip = [[0] * 10 for _ in range(10)]
        skip[1][3] = skip[3][1] = 2
        skip[1][7] = skip[7][1] = 4
        skip[3][9] = skip[9][3] = 6
        skip[7][9] = skip[9][7] = 8
        skip[1][9] = skip[9][1] = 5
        skip[2][8] = skip[8][2] = 5
        skip[3][7] = skip[7][3] = 5
        skip[4][6] = skip[6][4] = 5

        visited = [False] * 10

        def dfs(cur: int, length: int) -> int:
            if length > n:
                return 0
            count = 1 if length >= m else 0
            visited[cur] = True
            for nxt in range(1, 10):
                if not visited[nxt]:
                    mid = skip[cur][nxt]
                    if mid == 0 or visited[mid]:
                        count += dfs(nxt, length + 1)
            visited[cur] = False   # 回溯，撤销访问标记
            return count

        return dfs(1, 1) * 4 + dfs(2, 1) * 4 + dfs(5, 1)

复杂度

  • 时间：O(9!)（上界是 9 个键的排列数，板面固定，实际是常数级，约百万以内），对称剪枝后再降为约 1/3。
  • 空间：O(9)，递归深度最多 9 层，外加 visited 与 skip 两张常数大小表。
易错点与扩展（面试高频考点）
  1. 跨点表必须完整：漏掉 3-7、4-6 这类"穿过中心 5"的对称项会静默算多，务必把 8 组对称关系都补上。
  2. 回溯要撤销标记：visited[cur] = true 后递归，返回前必须置回 false，否则后续分支会误判已访问而漏解。
  3. 计数的位置：len >= m 时先 +1 再加子层结果（因为当前前缀本身就是合法手势），且 len > n 要提前返回。
  4. 对称系数别乘错：起点 1/2/5 的系数分别是 4 / 4 / 1，只算三次而不是九次。
  5. 可加记忆化：若把状态 (cur, visitedMask, len) 作为键做记忆化，可用位掩码压缩访问状态进一步加速，但本题规模小，回溯足够。

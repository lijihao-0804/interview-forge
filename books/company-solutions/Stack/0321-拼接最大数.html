题目与约束

给你两个整数数组 nums1 和 nums2，它们的长度分别为 m 和 n。数组 nums1 和 nums2 分别代表两个数各位上的数字。同时你也会得到一个整数 k。

请你利用这两个数组中的数字创建一个长度为 k <= m + n 的最大数。同一数组中数字的相对顺序必须保持不变。

返回代表答案的长度为 k 的数组。

示例 1：

输入：nums1 = [3,4,6,5], nums2 = [9,1,2,5,8,3], k = 5
输出：[9,8,6,5,3]

示例 2：

输入：nums1 = [6,7], nums2 = [6,0,4], k = 5
输出：[6,7,6,0,4]

示例 3：

输入：nums1 = [3,9], nums2 = [8,9], k = 3
输出：[9,8,9]

提示：

  • m == nums1.length
  • n == nums2.length
  • 1 <= m, n <= 500
  • 0 <= nums1[i], nums2[i] <= 9
  • 1 <= k <= m + n
  • nums1 和 nums2 没有前导 0

核心不变量

对每个"从 nums1 取 i 个、从 nums2 取 k-i 个"的合法分配，各自用单调栈求字典序最大的子序列，再按剩余部分的字典序贪心归并，所有分配中的最大者即为全局最优。

完整推导

思路推导

  1. 拆成两个子问题

最终结果由两部分拼成：从 nums1 里选 i 个、从 nums2 里选 k-i 个（i 从 max(0, k-n) 到 min(k, m)），且各自内部保持相对顺序。先解决"从一个数组里选 t 个数、保持顺序、字典序最大"这个子问题。

  1. 单调栈取最大子序列

要长度 t 最大：从左往右扫，用一个栈；若还能删（已删数 < len-t）且栈顶比当前数字小，就弹栈（把更小的前导数字删掉，让更大的数字顶上来）。扫完取前 t 个即可。例：nums=[3,4,6,5] 取 2 个，栈过程 3 → 4(弹3) → 6(弹4) → 6,5，得到 [6,5]。

  1. 归并时要"看全局"

把两个子序列合并成长度 k 的最大序列，不能简单比较当前两个数字：若相等要继续往后比，哪边剩余序列字典序更大就先取哪边的当前元素。例：[6,7] 与 [6,0,4] 比较时当前都是 6，需看后缀 7 vs 0，取前者。

  1. 枚举取最优

对所有合法 i 生成候选，逐一与当前最优比较，保留最大者。因为分配方案只有 O(k) 种，每种内部 O(m+n)，总复杂度 O(k·(m+n))，m,n ≤ 500 完全可承受。

Java 实现（单调栈 + 归并）

class Solution {
    public int[] maxNumber(int[] nums1, int[] nums2, int k) {
        int m = nums1.length, n = nums2.length;
        int[] best = new int[0];
        // i = 从 nums1 取的个数；保证两边都取得到合法数量
        for (int i = Math.max(0, k - n); i <= Math.min(k, m); i++) {
            int[] seq1 = maxSubsequence(nums1, i);
            int[] seq2 = maxSubsequence(nums2, k - i);
            int[] merged = merge(seq1, seq2);
            if (greater(merged, 0, best, 0)) {
                best = merged;
            }
        }
        return best;
    }

    /** 从 nums 中保留 t 个数字，保持相对顺序，使结果字典序最大（单调栈） */
    private int[] maxSubsequence(int[] nums, int t) {
        int drop = nums.length - t;   // 还可以删除的元素个数
        int[] stack = new int[t];
        int top = 0;
        for (int num : nums) {
            // 只要还能删，且栈顶比当前小，就弹栈让更大的数顶上来
            while (drop > 0 && top > 0 && stack[top - 1] < num) {
                top--;
                drop--;
            }
            if (top < t) {
                stack[top++] = num;   // 还没选满，入栈
            } else {
                drop--;               // 已选满，当前元素被丢弃
            }
        }
        return stack;
    }

    /** 合并两个序列，使结果字典序最大 */
    private int[] merge(int[] a, int[] b) {
        int[] res = new int[a.length + b.length];
        int i = 0, j = 0, idx = 0;
        while (i < a.length || j < b.length) {
            if (greater(a, i, b, j)) {
                res[idx++] = a[i++];
            } else {
                res[idx++] = b[j++];
            }
        }
        return res;
    }

    /** 比较 a[i..] 与 b[j..] 的字典序，a 更大返回 true（相等时 b 先耗尽也算 a 大） */
    private boolean greater(int[] a, int i, int[] b, int j) {
        while (i < a.length && j < b.length && a[i] == b[j]) {
            i++;
            j++;
        }
        return j == b.length || (i < a.length && a[i] > b[j]);
    }
}

复杂度

  • 时间：O(k·(m+n))。合法的分配 i 最多 min(k,m)+1 个；每个分配求两条子序列 O(m+n)、归并 O(k)。m,n ≤ 500，总体可接受。
  • 空间：O(m+n)。存放两条子序列、归并结果与单调栈，均可复用。

易错点与扩展（面试高频考点）

  1. 合并时不能只比当前位：当 a[i] == b[j] 时必须继续往后比较整个后缀的字典序，否则会因"短视"选错，例 2 是关键反例。
  2. 单调栈的删除配额 drop = len - t：栈满之后继续扫描时，多余元素要"丢掉"并消耗配额，否则长度会超过 t。
  3. 分配枚举的边界：i 的范围是 [max(0, k-n), min(k, m)]，越界会导致某一边无法凑够 k-i 或 i 个。
  4. 空结果比较：best 初始为空数组时，greater(cand, 0, best, 0) 中 j == b.length 直接返回 true，逻辑自洽。
  5. 扩展：本题是"移除 k 位数字使剩余最大"（LeetCode 402）的多数组版本，核心都是单调栈；若要求数字顺序也任意，则退化为直接排序取前 k 大，失去难度。

力扣原题

321. 拼接最大数

C++ 实现

#include <vector>
#include <algorithm>
using namespace std;

class Solution {
    // 从 nums 中保留 t 个数字，保持相对顺序，字典序最大
    vector<int> maxSubsequence(const vector<int>& nums, int t) {
        int drop = (int)nums.size() - t;
        vector<int> st;
        for (int x : nums) {
            while (drop > 0 && !st.empty() && st.back() < x) {
                st.pop_back();
                --drop;
            }
            if ((int)st.size() < t) {
                st.push_back(x);
            } else {
                --drop;
            }
        }
        return st;
    }

    // 比较 a[i..] 与 b[j..]，a 更大返回 true
    bool greater(const vector<int>& a, int i, const vector<int>& b, int j) {
        while (i < (int)a.size() && j < (int)b.size() && a[i] == b[j]) {
            ++i;
            ++j;
        }
        return j == (int)b.size() || (i < (int)a.size() && a[i] > b[j]);
    }

    vector<int> merge(const vector<int>& a, const vector<int>& b) {
        vector<int> res;
        int i = 0, j = 0;
        while (i < (int)a.size() || j < (int)b.size()) {
            if (greater(a, i, b, j)) res.push_back(a[i++]);
            else res.push_back(b[j++]);
        }
        return res;
    }

public:
    vector<int> maxNumber(vector<int>& nums1, vector<int>& nums2, int k) {
        int m = nums1.size(), n = nums2.size();
        vector<int> best;
        for (int i = max(0, k - n); i <= min(k, m); ++i) {
            vector<int> cand = merge(maxSubsequence(nums1, i), maxSubsequence(nums2, k - i));
            if (greater(cand, 0, best, 0)) best = cand;
        }
        return best;
    }
};

Python 实现

class Solution:
    def maxNumber(self, nums1: list[int], nums2: list[int], k: int) -> list[int]:
        def max_subsequence(nums, t):
            # 保留 t 个数字，字典序最大
            drop = len(nums) - t
            stack = []
            for x in nums:
                while drop > 0 and stack and stack[-1] < x:
                    stack.pop()
                    drop -= 1
                if len(stack) < t:
                    stack.append(x)
                else:
                    drop -= 1
            return stack

        def greater(a, i, b, j):
            # a[i:] 是否字典序大于 b[j:]
            while i < len(a) and j < len(b) and a[i] == b[j]:
                i += 1
                j += 1
            if j == len(b):
                return True
            if i == len(a):
                return False
            return a[i] > b[j]

        def merge(a, b):
            res = []
            i = j = 0
            while i < len(a) or j < len(b):
                if greater(a, i, b, j):
                    res.append(a[i])
                    i += 1
                else:
                    res.append(b[j])
                    j += 1
            return res

        best = []
        for i in range(max(0, k - len(nums2)), min(k, len(nums1)) + 1):
            cand = merge(max_subsequence(nums1, i), max_subsequence(nums2, k - i))
            if greater(cand, 0, best, 0):
                best = cand
        return best

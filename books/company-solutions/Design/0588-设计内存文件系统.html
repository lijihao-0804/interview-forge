核心不变量

每个目录 / 文件都是 Trie 上的一个节点：目录节点持有"名字 → 子节点"的有序映射，文件节点额外持有内容；一条斜杠分隔的路径就是从根逐段向下走。

完整推导

思路推导

  1. 用 Trie 表示目录层级

文件系统天然是树，路径 /a/b/c 就是一条从根出发的链。每个节点用 children: 名字 -> 子节点 记录下级，再加一个 isFile 标记与 content 字符串区分文件/目录。用有序 map（TreeMap / map / sorted） 存 children，ls 直接拿到字典序。

  1. 路径解析：逐段下钻

把 path 按 / 切分，跳过空串（开头 / 产生的空段），从根节点一段段 children.get 往下走，最终停在目标节点。

  1. 四个操作各自落地
  • ls：先 traverse 到节点；是文件就返回 [文件名]（取路径最后一段），是目录就返回 children 的有序键集合。
  • mkdir：逐段下钻，缺哪层建哪层（putIfAbsent）。
  • addContentToFile：逐段下钻，最后一段建为文件节点并 append 内容（已存在则追加）。
  • readContentFromFile：直接 traverse 取内容。
  1. 小例子（示例 1）

mkdir("/a/b/c") 建出 a→b→c。addContentToFile("/a/b/c/d","hello") 在 c 下建文件 d 写入内容。ls("/") 看根的有序 children，只有 a，返回 ["a"]。readContentFromFile("/a/b/c/d") 沿 a→b→c→d 拿到 "hello"。

Java 实现（FileSystem）

import java.util.*;

class FileSystem {
    // Trie 节点：目录或文件
    private static class Node {
        Map<String, Node> children = new TreeMap<>(); // 有序，保证 ls 字典序
        boolean isFile = false;
        StringBuilder content = new StringBuilder();
    }

    private final Node root;

    public FileSystem() {
        root = new Node();
    }

    // 沿路径下钻，返回目标节点（路径合法时一定存在）
    private Node traverse(String path) {
        Node cur = root;
        for (String part : path.split("/")) {
            if (part.isEmpty()) continue; // 跳过开头 / 产生的空段
            cur = cur.children.get(part);
        }
        return cur;
    }

    public List<String> ls(String path) {
        Node node = traverse(path);
        List<String> res = new ArrayList<>();
        if (node.isFile) {
            // 文件只返回自身名字
            String[] parts = path.split("/");
            res.add(parts[parts.length - 1]);
        } else {
            // 目录返回有序子项
            res.addAll(node.children.keySet());
        }
        return res;
    }

    public void mkdir(String path) {
        Node cur = root;
        for (String part : path.split("/")) {
            if (part.isEmpty()) continue;
            cur.children.putIfAbsent(part, new Node());
            cur = cur.children.get(part);
        }
    }

    public void addContentToFile(String filePath, String content) {
        String[] parts = filePath.split("/");
        Node cur = root;
        for (int i = 0; i < parts.length; i++) {
            if (parts[i].isEmpty()) continue;
            if (i == parts.length - 1) {
                // 最后一段是文件
                cur.children.putIfAbsent(parts[i], new Node());
                Node file = cur.children.get(parts[i]);
                file.isFile = true;
                file.content.append(content); // 已存在则追加
                return;
            }
            cur.children.putIfAbsent(parts[i], new Node());
            cur = cur.children.get(parts[i]);
        }
    }

    public String readContentFromFile(String filePath) {
        return traverse(filePath).content.toString();
    }
}

复杂度

  • 时间：单次操作 O(L + k log k)——L 为路径长度（逐段下钻），k 为目录条目数（有序 map 维护 / 返回顺序的成本）。mkdir、addContentToFile、readContentFromFile 为 O(L)。每个函数最多调用 100 次，完全够用。
  • 空间：O(S)，S 为所有路径名与文件内容的总字符数，即整棵 Trie 与内容的规模。

易错点与扩展（面试高频考点）

  1. 路径切分的空段："/a/b".split("/") 在 Java 中首段是空串，必须 if (part.isEmpty()) continue; 跳过，否则会在根下建出空名字节点。
  2. ls 对文件与目录的分支：文件返回 [文件名]（要取路径最后一段），目录才返回子项列表，二者不能混。
  3. addContentToFile 的追加语义：文件已存在时是 append 而不是覆盖，示例里多次写入同一文件就会体现。
  4. 字典序输出：用 TreeMap（C++ map、Python sorted）天然有序，比每次手动排序更省心。
  5. mkdir 要为中间目录补建：用 putIfAbsent 逐段建，缺哪层补哪层；不存在"已存在就报错"的语义。

C++ 实现

class FileSystem {
    struct Node {
        map<string, Node*> children; // 有序，保证 ls 字典序
        bool isFile = false;
        string content;
    };

    Node* root;

    Node* traverse(const string& path) {
        Node* cur = root;
        stringstream ss(path);
        string part;
        while (getline(ss, part, '/')) {
            if (part.empty()) continue;
            cur = cur->children[part];
        }
        return cur;
    }

public:
    FileSystem() { root = new Node(); }

    vector<string> ls(string path) {
        Node* node = traverse(path);
        vector<string> res;
        if (node->isFile) {
            int idx = path.find_last_of('/');
            res.push_back(path.substr(idx + 1));
        } else {
            for (auto& kv : node->children) res.push_back(kv.first);
        }
        return res;
    }

    void mkdir(string path) {
        Node* cur = root;
        stringstream ss(path);
        string part;
        while (getline(ss, part, '/')) {
            if (part.empty()) continue;
            if (!cur->children.count(part)) cur->children[part] = new Node();
            cur = cur->children[part];
        }
    }

    void addContentToFile(string filePath, string content) {
        vector<string> parts;
        stringstream ss(filePath);
        string part;
        while (getline(ss, part, '/')) {
            if (!part.empty()) parts.push_back(part);
        }
        Node* cur = root;
        for (int i = 0; i < (int)parts.size(); i++) {
            if (i == (int)parts.size() - 1) {
                if (!cur->children.count(parts[i])) cur->children[parts[i]] = new Node();
                Node* file = cur->children[parts[i]];
                file->isFile = true;
                file->content += content; // 已存在则追加
                return;
            }
            if (!cur->children.count(parts[i])) cur->children[parts[i]] = new Node();
            cur = cur->children[parts[i]];
        }
    }

    string readContentFromFile(string filePath) {
        return traverse(filePath)->content;
    }
};

Python 实现

class FileSystem:
    class Node:
        __slots__ = ("children", "is_file", "content")

        def __init__(self):
            self.children = {}
            self.is_file = False
            self.content = []

    def __init__(self):
        self.root = FileSystem.Node()

    def _traverse(self, path):
        cur = self.root
        for part in path.split("/"):
            if not part:
                continue
            cur = cur.children[part]
        return cur

    def ls(self, path):
        node = self._traverse(path)
        if node.is_file:
            return [path.split("/")[-1]]
        return sorted(node.children.keys())

    def mkdir(self, path):
        cur = self.root
        for part in path.split("/"):
            if not part:
                continue
            if part not in cur.children:
                cur.children[part] = FileSystem.Node()
            cur = cur.children[part]

    def addContentToFile(self, filePath, content):
        parts = [p for p in filePath.split("/") if p]
        cur = self.root
        for i, part in enumerate(parts):
            if i == len(parts) - 1:
                if part not in cur.children:
                    cur.children[part] = FileSystem.Node()
                node = cur.children[part]
                node.is_file = True
                node.content.append(content)  # 已存在则追加
                return
            if part not in cur.children:
                cur.children[part] = FileSystem.Node()
            cur = cur.children[part]

    def readContentFromFile(self, filePath):
        return "".join(self._traverse(filePath).content)

449. 序列化和反序列化二叉搜索树

题目与约束

序列化是将数据结构或对象转换为一系列位的过程，以便它可以存储在文件或内存缓冲区中，或通过网络连接链路传输，以便稍后在同一个或另一个计算机环境中重建。

设计一个算法来序列化和反序列化 二叉搜索树 。 对序列化/反序列化算法的工作方式没有限制。 您只需确保二叉搜索树可以序列化为字符串，并且可以将该字符串反序列化为最初的二叉搜索树。

编码的字符串应尽可能紧凑。

示例 1：

输入：root = [2,1,3]
输出：[2,1,3]

示例 2：

输入：root = []
输出：[]

提示：

  • 树中节点数范围是 [0, 104]

  • 0 <= Node.val <= 104

  • 题目数据 保证 输入的树是一棵二叉搜索树。

核心不变量

二叉搜索树的前序遍历序列足以唯一重建整棵树：处理序列中下一个值时，只要它落在当前节点允许的 (low, high) 区间内，它就是当前节点子树中的节点；否则该子树为空。

思路推导

  1. 直觉与痛点（普通二叉树需要空占位符）

一般的二叉树序列化（LeetCode 297）用前序遍历时必须把 null 也写进字符串，否则无法区分「左子树为空」还是「序列已结束」。写大量 null 占位符会让编码很臃肿，不符合本题「尽可能紧凑」的要求。

  1. 破局思路（BST 自带顺序信息）

BST 的中序遍历是升序。已知前序 + 中序可唯一重建二叉树，而中序对 BST 来说就是「排好序的前序」，其实不需要额外存储。于是可以只存前序遍历，反序列化时利用 BST 的「左小右大」上下界来判断每个值该挂在哪。

  1. 反序列化怎么用上下界

维护一个全局下标 idx 指向当前待处理的前序值。递归函数 build(low, high)： - 若 idx 越界，返回空； - 取 val = preorder[idx]，若 val <= low || val >= high，说明这个值不属于当前子树，立即返回空（不消耗 idx）； - 否则消耗它作为当前节点，递归 build(low, val) 建左子树、build(val, high) 建右子树。

例如前序 [2,1,3]：根 2 的区间 (-∞, +∞)；左子树区间 (-∞, 2)，值 1 合法成为左孩子，其左/右子区间都容不下 3，回溯；根 2 的右子树区间 (2, +∞)，值 3 合法成为右孩子。重建完成。

  1. 为什么紧凑

字符串里只有节点值，没有任何 null 占位符，长度与节点数成正比，达到「尽可能紧凑」。

Java 实现（前序 + 上下界）

/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode(int x) { val = x; }
 * }
 */
public class Codec {

    // Encodes a tree to a single string.
    public String serialize(TreeNode root) {
        StringBuilder sb = new StringBuilder();
        preorder(root, sb);
        return sb.toString();
    }

    private void preorder(TreeNode node, StringBuilder sb) {
        if (node == null) {
            return;
        }
        if (sb.length() > 0) {
            sb.append(',');
        }
        sb.append(node.val);
        preorder(node.left, sb);
        preorder(node.right, sb);
    }

    // Decodes your encoded data to tree.
    public TreeNode deserialize(String data) {
        if (data == null || data.isEmpty()) {
            return null;
        }
        String[] parts = data.split(",");
        int[] idx = {0}; // 用数组包装下标，便于递归中共享与修改
        return build(parts, idx, Long.MIN_VALUE, Long.MAX_VALUE);
    }

    private TreeNode build(String[] parts, int[] idx, long low, long high) {
        if (idx[0] >= parts.length) {
            return null;
        }
        int val = Integer.parseInt(parts[idx[0]]);
        // 不在合法上下界内：这个值属于祖先的其它子树，当前子树为空
        if (val <= low || val >= high) {
            return null;
        }
        idx[0]++; // 消耗当前值
        TreeNode node = new TreeNode(val);
        node.left = build(parts, idx, low, val);   // 左子树必须 < val
        node.right = build(parts, idx, val, high); // 右子树必须 > val
        return node;
    }
}

C++ 实现

#include <string>
#include <vector>
#include <sstream>
#include <climits>
using namespace std;

/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode(int x) : val(x), left(NULL), right(NULL) {}
 * };
 */
class Codec {
public:
    // Encodes a tree to a single string.
    string serialize(TreeNode* root) {
        string out;
        preorder(root, out);
        return out;
    }

    // Decodes your encoded data to tree.
    TreeNode* deserialize(string data) {
        if (data.empty()) return nullptr;
        vector<int> vals;
        stringstream ss(data);
        string token;
        while (getline(ss, token, ',')) {
            vals.push_back(stoi(token));
        }
        int idx = 0;
        return build(vals, idx, LONG_MIN, LONG_MAX);
    }

private:
    void preorder(TreeNode* node, string& out) {
        if (!node) return;
        if (!out.empty()) out += ',';
        out += to_string(node->val);
        preorder(node->left, out);
        preorder(node->right, out);
    }

    TreeNode* build(vector<int>& vals, int& idx, long low, long high) {
        if (idx >= (int)vals.size()) return nullptr;
        int val = vals[idx];
        if (val <= low || val >= high) return nullptr; // 不属于当前子树
        idx++;                                          // 消耗当前值
        TreeNode* node = new TreeNode(val);
        node->left = build(vals, idx, low, val);        // 左子树 < val
        node->right = build(vals, idx, val, high);      // 右子树 > val
        return node;
    }
};

Python 实现

# Definition for a binary tree node.
# class TreeNode:
#     def __init__(self, x):
#         self.val = x
#         self.left = None
#         self.right = None
from typing import Optional

class Codec:
    def serialize(self, root: Optional[TreeNode]) -> str:
        parts = []

        def preorder(node):
            if not node:
                return
            parts.append(str(node.val))
            preorder(node.left)
            preorder(node.right)

        preorder(root)
        return ",".join(parts)

    def deserialize(self, data: str) -> Optional[TreeNode]:
        if not data:
            return None
        vals = list(map(int, data.split(",")))
        self.idx = 0

        def build(low, high):
            if self.idx >= len(vals):
                return None
            val = vals[self.idx]
            if val <= low or val >= high:  # 不属于当前子树的区间
                return None
            self.idx += 1                  # 消耗当前值
            node = TreeNode(val)
            node.left = build(low, val)    # 左子树 < val
            node.right = build(val, high)  # 右子树 > val
            return node

        return build(float("-inf"), float("inf"))

复杂度

  • 时间：O(n)，序列化遍历每个节点一次；反序列化每个值最多被检查一次并消耗一次。
  • 空间：O(n)，序列化结果字符串与反序列化的递归栈（最坏链状树深度 n）均为线性。
易错点与扩展（面试高频考点）
  1. 上下界要用严格不等式：val <= low || val >= high 才判空，因为 BST 中不允许相等值，用 long 边界可避免 Integer.MIN/MAX 溢出。
  2. 反序列化不要「提前消耗」：判空时应先看值是否越界，越界就 return null 且不 idx++，否则会把本该属于祖先的值吞掉。
  3. 空树与单节点：空树序列化得到空串，反序列化要能正确返回 null；单节点 [1] 反序列化后左右子树都应为空。
  4. 面试追问——和 297 题（序列化普通二叉树）的区别：普通二叉树必须写 null 占位才能定形；本题借助 BST 特性省掉占位符，字符串更紧凑，正是题目强调的优化点。
  5. 扩展——用「前序 + 中序（排序）」或「后序」：也可以用后序遍历 + 上下界，自底向上重建；思路对称，代码稍作调整即可。
